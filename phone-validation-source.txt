"use client";
import HikingDatePicker, { type BookingCapacitySummary } from "./HikingDatePicker";

import { useEffect, useMemo, useState } from "react";
import { useRouter } from "next/navigation";
import { useSession } from "next-auth/react";
import {
  AlertCircle,
  ArrowLeft,
  CheckCircle2,
  LoaderCircle,
  Plus,
  Trash2,
  Users,
} from "lucide-react";

type ApiPackageService = {
  id: number;
  package_id: number;
  service: string;
};

type ApiPackage = {
  id: number;
  name: string;
  description: string;
  price: string | number;
  max_tourists: number;
  status: string;
  is_walk_in?: boolean;
  services: ApiPackageService[];
};

type ApiTrail = {
  id: number;
  name: string;
  description?: string | null;
  difficulty?: string | null;
  is_active: boolean | number;
};

type Tourist = {
  full_name: string;
  age: string;
  gender: string;
  nationality: string;
  contact_number: string;
  address: string;
  emergency_contact: string;
  emergency_number: string;
};

type ApiError = {
  message?: string;
  errors?: Record<string, string | string[]>;
};

const emptyTourist = (): Tourist => ({
  full_name: "",
  age: "",
  gender: "",
  nationality: "",
  contact_number: "",
  address: "",
  emergency_contact: "",
  emergency_number: "",
});

function localDateString(date: Date): string {
  const year = date.getFullYear();
  const month = String(date.getMonth() + 1).padStart(2, "0");
  const day = String(date.getDate()).padStart(2, "0");

  return `${year}-${month}-${day}`;
}

function getErrorMessage(
  data: ApiError,
  fallback: string,
): string {
  if (typeof data.message === "string") {
    if (data.errors) {
      const firstError = Object.values(data.errors)
        .flatMap((value) =>
          Array.isArray(value) ? value : [value],
        )
        .find((value) => typeof value === "string");

      if (firstError) {
        return firstError;
      }
    }

    return data.message;
  }

  if (data.errors) {
    const firstError = Object.values(data.errors)
      .flatMap((value) =>
        Array.isArray(value) ? value : [value],
      )
      .find((value) => typeof value === "string");

    if (firstError) {
      return firstError;
    }
  }

  return fallback;
}

async function readResponse(
  response: Response,
): Promise<Record<string, unknown>> {
  const text = await response.text();

  if (!text.trim()) {
    return {};
  }

  try {
    const parsed: unknown = JSON.parse(text);

    if (
      parsed &&
      typeof parsed === "object" &&
      !Array.isArray(parsed)
    ) {
      return parsed as Record<string, unknown>;
    }

    return {};
  } catch {
    throw new Error(
      "The server returned an unreadable response.",
    );
  }
}

function formatPrice(price: number): string {
  if (!Number.isFinite(price)) {
    return "Price unavailable";
  }

  return new Intl.NumberFormat("en-PH", {
    style: "currency",
    currency: "PHP",
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  }).format(price);
}

export default function BookPackage({
  initialPackageId,
}: {
  initialPackageId: string;
}) {
  const router = useRouter();
  const { data: session, status: sessionStatus } =
    useSession();

  const [packages, setPackages] = useState<ApiPackage[]>([]);
  const [trails, setTrails] = useState<ApiTrail[]>([]);
  const [packageId, setPackageId] = useState(initialPackageId);
  const [trailId, setTrailId] = useState("");
  const [hikingDate, setHikingDate] = useState("");
  const [tourists, setTourists] = useState<Tourist[]>([
    emptyTourist(),
  ]);
  const [remarks, setRemarks] = useState("");
  const [loading, setLoading] = useState(true);
  const [submitting, setSubmitting] = useState(false);
  const [error, setError] = useState("");
  const [success, setSuccess] = useState("");
  const [bookingNumber, setBookingNumber] = useState("");
  const [capacitySummary, setCapacitySummary] = useState<BookingCapacitySummary | null>(null);

  const apiBaseUrl =
    process.env.NEXT_PUBLIC_LARAVEL_API_URL ||
    "http://localhost:18000";

  const accessToken = session?.user?.accessToken;

  const selectedPackage = useMemo(
    () =>
      packages.find(
        (item) => String(item.id) === packageId,
      ),
    [packages, packageId],
  );

  const minimumHikingDate = useMemo(() => {
    const date = new Date();
    date.setHours(12, 0, 0, 0);
    date.setDate(date.getDate() + 30);

    return localDateString(date);
  }, []);


  const maximumTourists = selectedPackage?.is_walk_in ? 1 : Math.min(12, Math.max(1, Number(selectedPackage?.max_tourists) || 12));

  const totalPrice = Number(selectedPackage?.price) * tourists.length;

  useEffect(() => {
    setPackageId(initialPackageId);
  }, [initialPackageId]);

  useEffect(() => {
    if (sessionStatus === "loading") {
      setLoading(true);
      return;
    }

    if (!accessToken) {
      setLoading(false);
      setError("Please sign in before making a booking.");
      return;
    }

    let cancelled = false;

    async function loadFormOptions() {
      setLoading(true);
      setError("");

      try {
        const headers = {
          Accept: "application/json",
          Authorization: `Bearer ${accessToken}`,
        };

        const [packageResponse, trailResponse] =
          await Promise.all([
            fetch(`${apiBaseUrl}/api/packages`, {
              headers,
              cache: "no-store",
            }),
            fetch(`${apiBaseUrl}/api/trails`, {
              headers,
              cache: "no-store",
            }),
          ]);

        const [packageData, trailData] = await Promise.all([
          readResponse(packageResponse),
          readResponse(trailResponse),
        ]);

        if (!packageResponse.ok) {
          throw new Error(
            getErrorMessage(
              packageData as ApiError,
              "Unable to load packages.",
            ),
          );
        }

        if (!trailResponse.ok) {
          throw new Error(
            getErrorMessage(
              trailData as ApiError,
              "Unable to load active trails.",
            ),
          );
        }

        const packageList = Array.isArray(packageData.packages)
          ? (packageData.packages as ApiPackage[]).filter(
              (item) =>
                item.status?.toUpperCase() === "ACTIVE",
            )
          : [];

        const trailList = Array.isArray(trailData.trails)
          ? (trailData.trails as ApiTrail[]).filter(
              (item) =>
                item.is_active === true ||
                item.is_active === 1,
            )
          : [];

        if (cancelled) {
          return;
        }

        setPackages(packageList);
        setTrails(trailList);

        if (
          !packageList.some(
            (item) => String(item.id) === initialPackageId,
          )
        ) {
          setPackageId(
            packageList.length > 0
              ? String(packageList[0].id)
              : "",
          );
        }

        if (trailList.length > 0) {
          setTrailId(String(trailList[0].id));
        }
      } catch (err) {
        if (!cancelled) {
          setError(
            err instanceof Error
              ? err.message
              : "Unable to load booking options.",
          );
        }
      } finally {
        if (!cancelled) {
          setLoading(false);
        }
      }
    }

    void loadFormOptions();

    return () => {
      cancelled = true;
    };
  }, [
    accessToken,
    apiBaseUrl,
    initialPackageId,
    sessionStatus,
  ]);

  useEffect(() => {
    setTourists((current) => {
      if (current.length > maximumTourists) {
        return current.slice(0, maximumTourists);
      }

      return current.length > 0 ? current : [emptyTourist()];
    });
  }, [maximumTourists]);

  function updateTourist(
    index: number,
    field: keyof Tourist,
    value: string,
  ) {
    setTourists((current) =>
      current.map((tourist, touristIndex) =>
        touristIndex === index
          ? { ...tourist, [field]: value }
          : tourist,
      ),
    );
  }

  function addTourist() {
    setError("");

    if (tourists.length >= maximumTourists) {
      setError(
        `This package allows a maximum of ${maximumTourists} tourists per booking.`,
      );
      return;
    }

    setTourists((current) => [...current, emptyTourist()]);
  }

  function removeTourist(index: number) {
    if (tourists.length <= 1) {
      setError("A booking must include at least one tourist.");
      return;
    }

    setTourists((current) =>
      current.filter((_, touristIndex) => touristIndex !== index),
    );
    setError("");
  }

  async function submitBooking(
    event: React.FormEvent<HTMLFormElement>,
  ) {
    event.preventDefault();
    setError("");
    setSuccess("");
    setBookingNumber("");

    if (!accessToken) {
      setError("Your session has expired. Please sign in again.");
      return;
    }

    if (!selectedPackage) {
      setError("Please select an available package.");
      return;
    }

    if (!trails.some((trail) => String(trail.id) === trailId)) {
      setError("Please select an active trail.");
      return;
    }

    if (!hikingDate || hikingDate < minimumHikingDate) {
      setError("Please select a hiking date at least 30 days from today.");
      return;
    }

    if (!capacitySummary || capacitySummary.date !== hikingDate) {
      setError("Please wait for visitor capacity to load for the selected date.");
      return;
    }

    if (capacitySummary.is_blocked) {
      setError(
        capacitySummary.blocked_reason
          ? `This hiking date is unavailable: ${capacitySummary.blocked_reason}`
          : "This hiking date has been blocked by DENR and cannot be booked."
      );
      return;
    }

    if (capacitySummary.is_full || tourists.length > capacitySummary.remaining_capacity) {
      setError("There are not enough visitor slots remaining for your group on this hiking date.");
      return;
    }

    if (
      tourists.length < 1 ||
      tourists.length > maximumTourists
    ) {
      setError(
        `This booking must contain between 1 and ${maximumTourists} tourists.`,
      );
      return;
    }

    const incompleteIndex = tourists.findIndex(
      (tourist) =>
        !tourist.full_name.trim() ||
        !tourist.age ||
        Number(tourist.age) < (selectedPackage.is_walk_in ? 18 : 1) ||
        Number(tourist.age) > 150 ||
        !tourist.gender ||
        !tourist.nationality.trim() ||
        !tourist.emergency_contact.trim() ||
        !tourist.emergency_number.trim(),
    );

    if (incompleteIndex !== -1) {
      setError(
        `Please complete all required fields for Tourist ${incompleteIndex + 1}.`,
      );
      return;
    }

    setSubmitting(true);

    try {
      const response = await fetch(
        `${apiBaseUrl}/api/package-bookings`,
        {
          method: "POST",
          headers: {
            Accept: "application/json",
            "Content-Type": "application/json",
            Authorization: `Bearer ${accessToken}`,
          },
          body: JSON.stringify({
            package_id: Number(packageId),
            trail_id: Number(trailId),
            hiking_date: hikingDate,
            tourist_count: tourists.length,
            remarks: remarks.trim() || null,
            tourists: tourists.map((tourist) => ({
              full_name: tourist.full_name.trim(),
              age: Number(tourist.age),
              gender: tourist.gender,
              nationality: tourist.nationality.trim(),
              contact_number:
                tourist.contact_number.trim() || null,
              address: tourist.address.trim() || null,
              emergency_contact:
                tourist.emergency_contact.trim(),
              emergency_number:
                tourist.emergency_number.trim(),
            })),
          }),
        },
      );

      const data = await readResponse(response);

      if (!response.ok) {
        throw new Error(
          getErrorMessage(
            data as ApiError,
            "Unable to submit your booking.",
          ),
        );
      }

      const booking = data.booking as
        | { booking_number?: string }
        | undefined;

      setBookingNumber(booking?.booking_number || "");
      setSuccess(
        typeof data.message === "string"
          ? data.message
          : "Your booking has been submitted successfully.",
      );
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Unable to submit your booking. Please try again.",
      );
    } finally {
      setSubmitting(false);
    }
  }

  if (sessionStatus === "loading" || loading) {
    return (
      <div className="flex min-h-64 items-center justify-center gap-3 text-sm text-zinc-500">
        <LoaderCircle className="h-5 w-5 animate-spin" />
        Loading booking options...
      </div>
    );
  }

  if (success) {
    return (
      <section className="mx-auto max-w-2xl rounded-3xl border border-emerald-200 bg-white p-8 text-center shadow-sm dark:border-emerald-900 dark:bg-zinc-900">
        <div className="mx-auto flex h-16 w-16 items-center justify-center rounded-full bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-400">
          <CheckCircle2 className="h-9 w-9" />
        </div>

        <h1 className="mt-5 text-2xl font-bold text-zinc-900 dark:text-white">
          Booking submitted
        </h1>

        <p className="mt-3 text-sm leading-6 text-zinc-600 dark:text-zinc-300">
          {success}
        </p>

        {bookingNumber && (
          <div className="mt-5 rounded-xl bg-zinc-50 p-4 dark:bg-zinc-800">
            <p className="text-xs uppercase tracking-wider text-zinc-500">
              Booking number
            </p>
            <p className="mt-1 text-lg font-bold text-zinc-900 dark:text-white">
              {bookingNumber}
            </p>
          </div>
        )}

        <p className="mt-4 text-sm text-zinc-500 dark:text-zinc-400">
          Your booking is awaiting processing. Please follow the
          instructions from DENR regarding payment and requirements.
        </p>

        <div className="mt-7 flex flex-col justify-center gap-3 sm:flex-row">
          <button
            type="button"
            onClick={() => router.push("/user/reservations")}
            className="rounded-xl bg-emerald-700 px-5 py-3 text-sm font-semibold text-white hover:bg-emerald-800"
          >
            View My Reservations
          </button>

          <button
            type="button"
            onClick={() => router.push("/user")}
            className="rounded-xl border border-zinc-200 px-5 py-3 text-sm font-semibold text-zinc-700 hover:bg-zinc-50 dark:border-zinc-700 dark:text-zinc-200 dark:hover:bg-zinc-800"
          >
            Browse Packages
          </button>
        </div>
      </section>
    );
  }

  return (
    <div className="mx-auto max-w-4xl space-y-6">
      <button
        type="button"
        onClick={() => router.push("/user")}
        className="inline-flex items-center gap-2 text-sm font-semibold text-emerald-700 hover:text-emerald-800 dark:text-emerald-400"
      >
        <ArrowLeft className="h-4 w-4" />
        Back to packages
      </button>

      <section className="rounded-3xl bg-emerald-950 px-6 py-8 text-white sm:px-9">
        <p className="text-sm font-medium text-emerald-200">
          Mount Pulag National Park
        </p>
        <h1 className="mt-2 text-3xl font-bold">
          Book a Package
        </h1>
        <p className="mt-3 max-w-2xl text-sm leading-6 text-emerald-100/80">
          Select your package and hiking schedule, then provide the
          required details for every tourist in your booking.
        </p>
      </section>

      {error && (
        <div
          role="alert"
          className="flex items-start gap-3 rounded-2xl border border-red-200 bg-red-50 p-4 text-sm text-red-800 dark:border-red-900 dark:bg-red-950/30 dark:text-red-300"
        >
          <AlertCircle className="mt-0.5 h-5 w-5 shrink-0" />
          <p>{error}</p>
        </div>
      )}

      {packages.length === 0 || trails.length === 0 ? (
        <div className="rounded-2xl border border-amber-200 bg-amber-50 p-5 text-sm text-amber-900 dark:border-amber-900 dark:bg-amber-950/30 dark:text-amber-200">
          {packages.length === 0
            ? "There are no active packages available for booking."
            : "There are no active trails available for booking."}
          Please contact the administrator if you believe this is incorrect.
        </div>
      ) : (
        <form onSubmit={submitBooking} className="space-y-6">
          <section className="rounded-3xl border border-zinc-200 bg-white p-6 dark:border-zinc-800 dark:bg-zinc-900 sm:p-8">
            <h2 className="text-lg font-bold text-zinc-900 dark:text-white">
              1. Package and hiking schedule
            </h2>

            <div className="mt-5 grid gap-5 sm:grid-cols-2">
              <label className="block text-sm font-medium text-zinc-700 dark:text-zinc-300">
                Package <span className="text-red-500">*</span>
                <select
                  required
                  value={packageId}
                  onChange={(event) => setPackageId(event.target.value)}
                  className="mt-2 w-full rounded-xl border border-zinc-200 bg-white px-3 py-3 text-sm outline-none focus:border-emerald-600 dark:border-zinc-700 dark:bg-zinc-950 dark:text-white"
                >
                  <option value="">Select a package</option>
                  {packages.map((item) => (
                    <option key={item.id} value={item.id}>
                      {item.name}
                    </option>
                  ))}
                </select>
              </label>

              <label className="block text-sm font-medium text-zinc-700 dark:text-zinc-300">
                Active trail <span className="text-red-500">*</span>
                <select
                  required
                  value={trailId}
                  onChange={(event) => setTrailId(event.target.value)}
                  className="mt-2 w-full rounded-xl border border-zinc-200 bg-white px-3 py-3 text-sm outline-none focus:border-emerald-600 dark:border-zinc-700 dark:bg-zinc-950 dark:text-white"
                >
                  <option value="">Select a trail</option>
                  {trails.map((trail) => (
                    <option key={trail.id} value={trail.id}>
                      {trail.name}
                      {trail.difficulty
                        ? ` ÃƒÂ¢Ã¢â€šÂ¬Ã¢â‚¬Â ${trail.difficulty}`
                        : ""}
                    </option>
                  ))}
                </select>
              </label>

            <HikingDatePicker
              date={hikingDate}
              minimumDate={minimumHikingDate}
              apiBaseUrl={apiBaseUrl}
              accessToken={accessToken}
              visitorCount={tourists.length}
              onDateChange={setHikingDate}
              onCapacityChange={setCapacitySummary}
            />
            </div>

            {selectedPackage && (
              <div className="mt-6 rounded-2xl bg-emerald-50 p-5 dark:bg-emerald-950/30">
                <div className="flex flex-col justify-between gap-4 sm:flex-row sm:items-start">
                  <div>
                    <p className="font-semibold text-emerald-950 dark:text-emerald-100">
                      {selectedPackage.name}
                    </p>
                    <p className="mt-1 text-sm leading-6 text-emerald-900/70 dark:text-emerald-200/70">
                      {selectedPackage.description ||
                        "No package description provided."}
                    </p>
                  </div>

                  <div className="shrink-0 sm:text-right">
                    <p className="text-xs text-emerald-800/70 dark:text-emerald-200/70">
                      Package price per person
                    </p>
                    <p className="mt-1 text-xl font-bold text-emerald-950 dark:text-emerald-100">
                      {formatPrice(Number(selectedPackage.price))}
                    </p>

                          <CheckCircle2 className="mt-0.5 h-4 w-4 shrink-0" />
                          {service.service}
                        </li>

          <section className="rounded-3xl border border-zinc-200 bg-white p-6 dark:border-zinc-800 dark:bg-zinc-900 sm:p-8">
            <div className="flex flex-col justify-between gap-3 sm:flex-row sm:items-center">
              <div>
                <h2 className="text-lg font-bold text-zinc-900 dark:text-white">
                  2. Tourist information
                </h2>
                <p className="mt-1 text-sm text-zinc-500 dark:text-zinc-400">
                  Enter each tourist separately. Emergency contact
                  details are required for every tourist.
                </p>
              </div>

              <div className="inline-flex items-center gap-2 text-sm font-semibold text-zinc-600 dark:text-zinc-300">
                <Users className="h-4 w-4" />
                {tourists.length} / {maximumTourists}
              </div>
            </div>

            <div className="mt-6 space-y-6">
              {tourists.map((tourist, index) => (
                <fieldset
                  key={index}
                  className="rounded-2xl border border-zinc-200 p-4 dark:border-zinc-700 sm:p-5"
                >
                  <div className="flex items-center justify-between gap-3">
                    <legend className="px-1 text-base font-semibold text-zinc-900 dark:text-white">
                      Tourist {index + 1}
                    </legend>

                    {tourists.length > 1 && (
                      <button
                        type="button"
                        onClick={() => removeTourist(index)}
                        className="inline-flex items-center gap-1.5 rounded-lg px-2 py-1.5 text-xs font-semibold text-red-600 hover:bg-red-50 dark:text-red-400 dark:hover:bg-red-950/40"
                      >
                        <Trash2 className="h-3.5 w-3.5" />
                        Remove
                      </button>
                    )}
                  </div>

                  <div className="mt-4 grid gap-4 sm:grid-cols-2">
                    <label className="text-sm font-medium text-zinc-700 dark:text-zinc-300 sm:col-span-2">
                      Full name <span className="text-red-500">*</span>
                      <input
                        required
                        maxLength={255}
                        value={tourist.full_name}
                        onChange={(event) =>
                          updateTourist(
                            index,
                            "full_name",
                            event.target.value,
                          )
                        }
                        placeholder="Enter full name"
                        className="mt-2 w-full rounded-xl border border-zinc-200 bg-white px-3 py-3 text-sm outline-none focus:border-emerald-600 dark:border-zinc-700 dark:bg-zinc-950 dark:text-white"
                      />
                    </label>

                    <label className="text-sm font-medium text-zinc-700 dark:text-zinc-300">
                      Age <span className="text-red-500">*</span>
                      <input
                        required
                        type="number"
                        min={selectedPackage?.is_walk_in ? 18 : 1}
                        max={150}
                        value={tourist.age}
                        onChange={(event) =>
                          updateTourist(index, "age", event.target.value)
                        }
                        placeholder="Age"
                        className="mt-2 w-full rounded-xl border border-zinc-200 bg-white px-3 py-3 text-sm outline-none focus:border-emerald-600 dark:border-zinc-700 dark:bg-zinc-950 dark:text-white"
                      />
                    </label>

                    <label className="text-sm font-medium text-zinc-700 dark:text-zinc-300">
                      Gender <span className="text-red-500">*</span>
                      <select
                        required
                        value={tourist.gender}
                        onChange={(event) =>
                          updateTourist(index, "gender", event.target.value)
                        }
                        className="mt-2 w-full rounded-xl border border-zinc-200 bg-white px-3 py-3 text-sm outline-none focus:border-emerald-600 dark:border-zinc-700 dark:bg-zinc-950 dark:text-white"
                      >
                        <option value="">Select gender</option>
                        <option value="Male">Male</option>
                        <option value="Female">Female</option>
                        <option value="Other">Other</option>
                        <option value="Prefer not to say">
                          Prefer not to say
                        </option>
                      </select>
                    </label>

                    <label className="text-sm font-medium text-zinc-700 dark:text-zinc-300">
                      Nationality <span className="text-red-500">*</span>
                      <input
                        required
                        maxLength={255}
                        value={tourist.nationality}
                        onChange={(event) =>
                          updateTourist(
                            index,
                            "nationality",

function getErrorMessage(data: ApiResponse, fallback: string) {
  const firstError = Object.values(data.errors ?? {})
    .flatMap((value) => (Array.isArray(value) ? value : [value]))
    .find((value): value is string => typeof value === "string");
  return firstError || data.message || fallback;
}

    /** Start a direct USER <-> ORGANIZER conversation. No booking is required. */
    public function startDirect(Request $request): JsonResponse
    {
        $sender = $request->user();
        $role = strtoupper((string) $sender->role);

        if (!in_array($role, ['USER', 'ORGANIZER'], true)) {
            return response()->json([
                'message' => 'Only Tourist and Organizer accounts can start direct chats.',
            ], 403);
        }

        $validated = $request->validate([
            'recipient_id' => ['required', 'integer', 'exists:users,id'],
            'body' => ['required', 'string', 'max:5000'],
        ]);

        $targetRole = $role === 'USER' ? 'ORGANIZER' : 'USER';

        $recipient = User::query()
            ->whereKey($validated['recipient_id'])
            ->where('role', $targetRole)
            ->where('status', 'ACTIVE')
            ->first();

        if (!$recipient) {
            return response()->json([
                'message' => 'You can only message an active account of the opposite role.',
            ], 422);
        }

        $conversation = DB::transaction(function () use ($sender, $recipient, $validated) {
            $conversation = Conversation::query()
                ->where('type', 'DIRECT_MESSAGE')
                ->where(function ($q) use ($sender, $recipient) {
                    $q->where(function ($pair) use ($sender, $recipient) {
                        $pair->where('user_id', $sender->id)
                            ->where('recipient_id', $recipient->id);
                    })->orWhere(function ($pair) use ($sender, $recipient) {
                        $pair->where('user_id', $recipient->id)
                            ->where('recipient_id', $sender->id);
                    });
                })
                ->lockForUpdate()
                ->first();

            if (!$conversation) {
                $conversation = Conversation::create([
                    'user_id' => $sender->id,
                    'recipient_id' => $recipient->id,
                    'type' => 'DIRECT_MESSAGE',
                    'subject' => 'Direct message',
                    'status' => 'OPEN',
                    'last_message_at' => now('Asia/Manila'),
                ]);
            }

            if ($role !== 'ORGANIZER') {
                return response()->json([
                    'message' => 'Only organizer accounts can start package-booking conversations.',
                ], 403);
            }

            return $this->createPackageBookingConversation($request);
        }

        if ($type === 'ORGANIZER_SUPPORT') {
            if ($role !== 'ORGANIZER') {
                return response()->json([
                    'message' => 'Only organizer accounts can start DENR support conversations.',
                ], 403);
            }

            return $this->createOrganizerSupportConversation($request);
        }

        if ($role !== 'USER') {
            return response()->json([
                'message' => 'Only visitor accounts can start these requests.',
            ], 403);
        }

        if ($type === 'ORGANIZER_APPLICATION') {
            return $this->createOrganizerApplication($request);
        }

        if ($type === 'RESERVATION_REQUEST') {
            return $this->createReservationConversation($request);
        }

        throw ValidationException::withMessages([
            'type' => ['Select a valid request type.'],
        ]);
    }

    /**
     * Create or continue a conversation between the package organizer and customer.
     * An organizer may only access bookings belonging to their own package.
     */
    private function createPackageBookingConversation(Request $request): JsonResponse
    {
        $validated = $request->validate([
            'package_booking_id' => [
                'required',
                'integer',
                'exists:package_bookings,id',
            ],
            'body' => ['required', 'string', 'max:5000'],
        ]);

        $organizer = $request->user();

        $result = DB::transaction(function () use ($organizer, $validated) {
            $booking = PackageBooking::query()
                ->whereKey($validated['package_booking_id'])
                ->lockForUpdate()
                ->first();

            if (!$booking) {
                abort(404, 'Package booking not found.');
            }

            $booking->load(['package', 'customer']);

            $package = $booking->package;

            if (!$package || (int) $package->organizer_id !== (int) $organizer->id) {
                abort(404, 'Package booking not found.');
            }

            $customer = $booking->customer;

            if (!$customer) {
                abort(404, 'Package booking customer not found.');
            }

            $conversation = Conversation::query()
                ->where('package_booking_id', $booking->id)
                ->lockForUpdate()
                ->first();

            $created = false;

            if (!$conversation) {
                $bookingReference = $booking->booking_number
                    ? " {$booking->booking_number}"
                    : " #{$booking->id}";

                $conversation = Conversation::create([
                    'user_id' => $customer->id,
                    'organizer_id' => $organizer->id,
                    'package_booking_id' => $booking->id,
                    'type' => 'PACKAGE_BOOKING',
                    'subject' => 'Package Booking' . $bookingReference,
                    'status' => 'OPEN',
                    'last_message_at' => now('Asia/Manila'),
                ]);

                $created = true;
            } elseif (
                (int) $conversation->organizer_id !== (int) $organizer->id
                || (int) $conversation->user_id !== (int) $customer->id
            ) {
                abort(404, 'Package booking conversation not found.');
            }

            Message::create([
                'conversation_id' => $conversation->id,
                'sender_id' => $organizer->id,
                'body' => trim($validated['body']),
            ]);

            $conversation->update([
                'last_message_at' => now('Asia/Manila'),
                'status' => 'OPEN',
            ]);

            return [
                'conversation' => $conversation,
                'customer' => $customer,
                'created' => $created,
            ];
        });

        $conversation = $result['conversation'];
        $customer = $result['customer'];

        $customer->notify(new SystemNotification(
            'New Package Booking Message',
            "{$organizer->first_name} {$organizer->last_name} sent you a message about your package booking.",
            'conversation',
            null,
            ['conversation_id' => $conversation->id]
        ));

        return response()->json([
            'message' => $result['created']
                ? 'Package booking conversation started.'
                : 'Message sent successfully.',
            'conversation' => $conversation->fresh([
                'user:id,first_name,last_name,email,phone,role',
                'organizer:id,first_name,last_name,email,role',
                'packageBooking',
                'messages.sender:id,first_name,last_name,role',
            ]),
        ], $result['created'] ? 201 : 200);
    }

    public function sendMessage(Request $request, Conversation $conversation): JsonResponse
    {
        if (!$this->canAccess($request, $conversation)) {
            return response()->json([
                'message' => 'Conversation not found.',
            ], 404);
        }

        if ($conversation->status === 'CLOSED') {
            return response()->json([
                'message' => 'This conversation is closed and no longer accepts replies.',
            ], 422);
        }

        $validated = $request->validate([
            'body' => [
                'nullable',
                'string',
                'max:5000',
                'required_without_all:dot_certificate,business_permit',
            ],
            'dot_certificate' => [
                'nullable',
                'image',
                'mimes:jpg,jpeg,png,webp',
                'max:3072',
            ],
            'business_permit' => [
                'nullable',
                'image',
                'mimes:jpg,jpeg,png,webp',
                'max:3072',
            ],
        ]);

        if ($conversation->type !== 'ORGANIZER_APPLICATION' && (
            $request->hasFile('dot_certificate')
            || $request->hasFile('business_permit')
        )) {
            return response()->json([
                'message' => 'Document attachments are only accepted for organizer applications.',
            ], 422);
        }

        $message = DB::transaction(function () use ($request, $conversation, $validated) {
            $message = Message::create([
                'conversation_id' => $conversation->id,
                'sender_id' => $request->user()->id,
                'body' => isset($validated['body'])
                    ? trim($validated['body'])
                    : null,
            ]);

            $this->storeApplicationAttachments($message, $request);

            $conversation->update([
                'last_message_at' => now('Asia/Manila'),
                'status' => 'OPEN',
            ]);

            return $message;
        });

        $this->notifyConversationParticipants(
            $conversation->fresh(['user', 'organizer', 'recipient']),
            $request->user()
        );

        return response()->json([
            'message' => 'Message sent successfully.',
            'message_record' => $message->load([
                'sender:id,first_name,last_name,role',
                'attachments:id,message_id,document_type,original_name,mime_type,file_size',
            ]),
        ], 201);
    }

    /** Only ADMIN can grant the ORGANIZER role. */
    public function reviewOrganizerApplication(
        Request $request,
        Conversation $conversation,
        ActivityLogger $logger
    ): JsonResponse {
        if ($conversation->type !== 'ORGANIZER_APPLICATION') {
            return response()->json([
                'message' => 'This is not an organizer application.',
            ], 422);
        }

        $validated = $request->validate([
            'decision' => ['required', 'string', 'in:APPROVED,REJECTED,NEEDS_REVISION'],
            'review_notes' => [
                'nullable',
                'string',
                'max:2000',
                'required_if:decision,REJECTED,NEEDS_REVISION',
            ],
        ]);

        $application = $conversation->organizerApplication;

        if (!$application || $application->status !== 'PENDING') {
            return response()->json([
                'message' => 'Only pending organizer applications can be reviewed.',
            ], 422);
        }

        $decision = $validated['decision'];
        $admin = $request->user();

        DB::transaction(function () use ($application, $conversation, $decision, $validated, $admin) {
            $lockedApplication = OrganizerApplication::query()
                ->whereKey($application->id)
                ->lockForUpdate()
                ->firstOrFail();

            if ($lockedApplication->status !== 'PENDING') {
                abort(409, 'This application has already been reviewed. Refresh the conversation.');
            }

            $lockedApplication->update([
                'status' => $decision,
                'review_notes' => $validated['review_notes'] ?? null,
                'reviewed_by' => $admin->id,
                'reviewed_at' => now('Asia/Manila'),
            ]);

            if ($decision === 'APPROVED') {
                User::query()
                    ->whereKey($lockedApplication->user_id)
                    ->update([
                        'role' => 'ORGANIZER',
                    ]);

                $conversation->update(['status' => 'CLOSED']);
            } elseif ($decision === 'REJECTED') {
                // Rejected applicants remain USER accounts and may start a new application later.
                $conversation->update(['status' => 'CLOSED']);
            } else {
                $conversation->update(['status' => 'OPEN']);
            }
        });

        $application->loadMissing('user');
        $applicant = $application->user;

        $title = match ($decision) {
            'APPROVED' => 'Organizer Application Approved',
            'REJECTED' => 'Organizer Application Rejected',
            default => 'Organizer Application Needs Revision',
        };

        $messageText = match ($decision) {
            'APPROVED' => 'Your organizer application has been approved. Sign out and sign in again to access the Organizer Portal.',
            'REJECTED' => 'Your organizer application was not approved. You remain a USER account. Review the DENR feedback for details.',
            default => 'DENR needs revised information or clearer documents before your application can be processed.',
        };

        if ($applicant) {
            $applicant->notify(new SystemNotification(
                $title,
                $messageText,
                'organizer_application',
                null,
                [
                    'conversation_id' => $conversation->id,
                    'application_status' => $decision,
                    'review_notes' => $validated['review_notes'] ?? null,
                ]
            ));
        }

        $logger->log(
            $admin,
            'REVIEW_ORGANIZER_APPLICATION',
            'ORGANIZER_APPLICATION',
            "Organizer application #{$application->id} was marked {$decision}.",
            $request
        );

        return response()->json([
            'message' => $title . '.',
            'conversation' => $conversation->fresh([
                'user:id,first_name,last_name,email,phone,role',
                'organizerApplication.reviewer:id,first_name,last_name',
            ]),
        ]);
    }

    /**
     * Staff-only final step for reservation coordination.
     * The conversation itself never creates a reservation.
     */
    public function createOfficialReservation(
        Request $request,
        Conversation $conversation,
        BookingCapacityService $capacityService,
        ActivityLogger $logger
    ): JsonResponse {
        if ($conversation->type !== 'RESERVATION_REQUEST') {
            return response()->json([
                'message' => 'This conversation is not a reservation coordination request.',
            ], 422);
        }

        if ($conversation->created_reservation_id) {
            return response()->json([
                'message' => 'An official reservation has already been created for this conversation.',
                'reservation_id' => $conversation->created_reservation_id,
            ], 409);
        }

        if ($conversation->status === 'CLOSED') {
            return response()->json([
                'message' => 'This conversation is closed.',
            ], 422);
        }

        $validated = $request->validate([
            'group_leader_name' => ['required', 'string', 'max:255'],
            'group_leader_phone' => ['required', 'string', 'max:50'],
            'remarks' => ['nullable', 'string', 'max:2000'],
            'participants' => ['required', 'array', 'min:1', 'max:12'],
            'participants.*.full_name' => ['required', 'string', 'max:255'],
            'participants.*.age' => ['required', 'integer', 'min:1', 'max:120'],
            'participants.*.gender' => ['required', 'in:MALE,FEMALE'],
            'participants.*.nationality' => ['required', 'string', 'max:100'],
            'participants.*.contact_number' => ['nullable', 'string', 'max:50'],
            'participants.*.address' => ['nullable', 'string'],
            'participants.*.emergency_contact' => ['required', 'string', 'max:255'],
            'participants.*.emergency_number' => ['required', 'string', 'max:50'],
        ]);

        $customer = $conversation->user;

        if (!$customer || strtoupper((string) $customer->status) !== 'ACTIVE') {
            return response()->json([
                'message' => 'The visitor account is not active.',
            ], 422);
        }

        $trail = Trail::query()
            ->whereKey($conversation->requested_trail_id)
            ->where('is_active', true)
            ->first();

        if (!$trail) {
            return response()->json([
                'message' => 'The requested trail is no longer available.',
            ], 422);
        }

        $hikingDate = Carbon::parse(
            $conversation->requested_hiking_date,
            'Asia/Manila'
        )->startOfDay();

        $minimumDate = Carbon::today('Asia/Manila')->addDays(30);

        if ($hikingDate->lt($minimumDate)) {
            return response()->json([
                'message' => 'The requested hiking date must be at least 30 days from today.',
                'minimum_hiking_date' => $minimumDate->toDateString(),
            ], 422);
        }

        $blockedDate = BlockedDate::query()
            ->whereDate('date', $hikingDate->toDateString())
            ->first();

        if ($blockedDate) {
            return response()->json([
                'message' => 'The requested hiking date is unavailable.',
                'reason' => $blockedDate->reason,
            ], 422);
        }

        $participantCount = count($validated['participants']);

        if ($participantCount !== (int) $conversation->requested_visitors) {
            return response()->json([
                'message' => 'The participant list must match the group size recorded in the coordination request.',
                'requested_visitors' => (int) $conversation->requested_visitors,
                'submitted_participants' => $participantCount,
            ], 422);
        }

        $result = DB::transaction(function () use (
            $conversation,
            $validated,
            $participantCount,
            $trail,
            $customer,
            $hikingDate,
            $capacityService,
            $request
        ) {
            $lockedConversation = Conversation::query()
                ->whereKey($conversation->id)
                ->lockForUpdate()
                ->firstOrFail();

            if ($lockedConversation->created_reservation_id) {
                return ['conflict' => true];
            }

            $capacityService->lockForCapacityCheck();
            $summary = $capacityService->getSummary($hikingDate->toDateString());

            if (
                $summary['is_blocked']
                || $summary['current_capacity'] + $participantCount > BookingCapacityService::MAXIMUM_CAPACITY
            ) {
                return [
                    'capacity_error' => true,
                    'summary' => $summary,
                ];
            }

            $reservationNumber = 'EP-' . strtoupper(Str::random(8));

            while (Reservation::query()->where('reservation_number', $reservationNumber)->exists()) {
                $reservationNumber = 'EP-' . strtoupper(Str::random(8));
            }

            $reservation = Reservation::create([
                'reservation_number' => $reservationNumber,
                'group_leader_name' => trim($validated['group_leader_name']),
                'group_leader_phone' => trim($validated['group_leader_phone']),
                'hiking_date' => $hikingDate->toDateString(),
                'total_participants' => $participantCount,
                // Staff explicitly creates this record after confirming the slot.
                'status' => 'APPROVED',
                'remarks' => $validated['remarks']
                    ?? 'Recorded by DENR staff after reservation coordination.',
                'user_id' => $customer->id,
                'trail_id' => $trail->id,
            ]);

            $reservation->participants()->createMany($validated['participants']);

            $lockedConversation->update([
                'created_reservation_id' => $reservation->id,
                'reservation_created_at' => now('Asia/Manila'),
                'status' => 'CLOSED',
                'last_message_at' => now('Asia/Manila'),
            ]);

            Message::create([
                'conversation_id' => $lockedConversation->id,
                'sender_id' => $request->user()->id,
                'body' => "DENR recorded and approved your reservation {$reservation->reservation_number} for {$hikingDate->toDateString()}. This is now part of the official booking records.",
            ]);

            return ['reservation' => $reservation];
        });

        if (!empty($result['conflict'])) {
            return response()->json([
                'message' => 'An official reservation has already been created for this conversation.',
            ], 409);
        }

        if (!empty($result['capacity_error'])) {
            $summary = $result['summary'];

            return response()->json([
                'message' => $summary['is_blocked']
                    ? 'The requested hiking date is unavailable.'
                    : 'There is not enough remaining visitor capacity for this group.',
                'date' => $summary['date'],
                'current_capacity' => $summary['current_capacity'],
                'maximum_capacity' => $summary['maximum_capacity'],
                'remaining_capacity' => $summary['remaining_capacity'],
                'requested_visitors' => $participantCount,
                'blocked_reason' => $summary['blocked_reason'],
            ], 422);
        }

        $reservation = $result['reservation']->load([
            'trail',
            'participants',
            'user',
        ]);

        $customer->notify(new SystemNotification(
            'Reservation Recorded by DENR',
            "Your reservation {$reservation->reservation_number} has been recorded and approved by DENR staff.",
            'reservation',
            null,
            [
                'reservation_id' => $reservation->id,
                'reservation_number' => $reservation->reservation_number,
                'hiking_date' => $reservation->hiking_date,
                'trail_name' => $trail->name,
                'status' => 'APPROVED',
            ]
        ));

        $logger->log(
            $request->user(),
            'CREATE_COORDINATED_RESERVATION',
            'RESERVATION',
            "Created approved reservation {$reservation->reservation_number} from conversation #{$conversation->id}.",
            $request
        );

        return response()->json([
            'message' => 'Official reservation created and approved successfully.',
            'reservation' => $reservation,
            'conversation' => $conversation->fresh(['createdReservation']),
        ], 201);
    }

    private function createReservationConversation(Request $request): JsonResponse
    {
        $validated = $request->validate([
            'requested_hiking_date' => [
                'required',
                'date',
                'after_or_equal:' . Carbon::today('Asia/Manila')->addDays(30)->toDateString(),
            ],
            'requested_trail_id' => ['required', 'integer', 'exists:trails,id'],
            'requested_visitors' => ['required', 'integer', 'min:1', 'max:12'],
            'body' => ['required', 'string', 'max:5000'],
        ]);

        $trail = Trail::query()
            ->whereKey($validated['requested_trail_id'])
            ->where('is_active', true)
            ->first();

        if (!$trail) {
            return response()->json([
                'message' => 'Select an active trail.',
            ], 422);
        }

        $conversation = DB::transaction(function () use ($request, $validated) {
            $conversation = Conversation::create([
                'user_id' => $request->user()->id,
                'type' => 'RESERVATION_REQUEST',
                'subject' => 'Reservation Slot Coordination',
                'status' => 'OPEN',
                'requested_hiking_date' => $validated['requested_hiking_date'],
                'requested_trail_id' => $validated['requested_trail_id'],
                'requested_visitors' => $validated['requested_visitors'],
                'last_message_at' => now('Asia/Manila'),
            ]);

            Message::create([
                'conversation_id' => $conversation->id,
                'sender_id' => $request->user()->id,
                'body' => trim($validated['body']),
            ]);

            return $conversation;
        });

        $user = $request->user();

        $this->notifyActiveStaff(
            'New Reservation Coordination Request',
            "{$user->first_name} {$user->last_name} requested reservation-slot coordination for {$validated['requested_hiking_date']} on {$trail->name}.",
            [
                'conversation_id' => $conversation->id,
                'requested_hiking_date' => $validated['requested_hiking_date'],
                'requested_trail_id' => $trail->id,
                'requested_visitors' => $validated['requested_visitors'],
            ]
        );

        return response()->json([
            'message' => 'Reservation coordination request sent. This message does not confirm or create a reservation.',
            'conversation' => $conversation->fresh([
                'requestedTrail:id,name',
                'messages.sender:id,first_name,last_name,role',
            ]),
        ], 201);
    }

    private function storeApplicationAttachments(Message $message, Request $request): void
    {
        foreach ([
            'dot_certificate' => 'DOT_CERTIFICATE',
            'business_permit' => 'BUSINESS_PERMIT',
        ] as $field => $documentType) {
            if (!$request->hasFile($field)) {
                continue;
            }

            $file = $request->file($field);

            if (!$file || !$file->isValid()) {
                throw ValidationException::withMessages([
                    $field => ['The uploaded file could not be read. Please upload it again.'],
                ]);
            }

            if ($file->getSize() > self::MAX_ATTACHMENT_BYTES) {
                throw ValidationException::withMessages([
                    $field => ['Each document image must be 3 MB or smaller.'],
                ]);
            }

            MessageAttachment::create([
                'message_id' => $message->id,
                'document_type' => $documentType,
                'original_name' => substr($file->getClientOriginalName(), 0, 255),
                'mime_type' => $file->getMimeType() ?: 'application/octet-stream',
                'file_size' => (int) $file->getSize(),
                'file_content' => file_get_contents($file->getRealPath()),
            ]);
        }
    }

    private function canAccess(Request $request, Conversation $conversation): bool
    {
        $user = $request->user();
        $role = strtoupper((string) $user->role);
        $isStaff = in_array($role, ['ADMIN', 'EMPLOYEE'], true);

        // Direct messages are private to their USER and ORGANIZER participants.
        if (strtoupper((string) $conversation->type) === 'DIRECT_MESSAGE') {
            return !$isStaff
                && in_array($role, ['USER', 'ORGANIZER'], true)
                && (
                    (int) $conversation->user_id === (int) $user->id
                    || (int) $conversation->recipient_id === (int) $user->id
                );
        }

        // Staff retain access to the official DENR conversation workflows.
        return $isStaff
            || (int) $conversation->user_id === (int) $user->id
            || (int) $conversation->organizer_id === (int) $user->id
            || (int) $conversation->recipient_id === (int) $user->id;
    }

    private function notifyActiveStaff(string $title, string $message, array $data): void
    {
        User::query()
            ->whereIn('role', ['ADMIN', 'EMPLOYEE'])
            ->where('status', 'ACTIVE')
            ->get()
            ->each(function (User $staff) use ($title, $message, $data) {
                $staff->notify(new SystemNotification(
                    $title,
                    $message,
                    'conversation',
                    null,
                    $data
                ));
            });
    }

    private function notifyConversationParticipants(
        Conversation $conversation,
        User $sender
    ): void {
        if ($conversation->type === 'DIRECT_MESSAGE') {
            $recipientId = (int) $conversation->user_id === (int) $sender->id
                ? $conversation->recipient_id
                : $conversation->user_id;

            User::query()->find($recipientId)?->notify(new SystemNotification(
                'New Chatbox Message',
                "{$sender->first_name} {$sender->last_name} sent you a message.",
                'conversation',
                null,
                ['conversation_id' => $conversation->id]
            ));

            return;
        }

        if (in_array(strtoupper((string) $sender->role), ['ADMIN', 'EMPLOYEE'], true)) {
            $conversation->user?->notify(new SystemNotification(
                'New DENR Message',
                'DENR staff sent a new message in your E-PULAG conversation.',
                'conversation',
                null,
                ['conversation_id' => $conversation->id]
            ));

            return;
        }

        if ($conversation->type === 'PACKAGE_BOOKING') {
            if ((int) $sender->id === (int) $conversation->organizer_id) {
                $conversation->user?->notify(new SystemNotification(
                    'New Package Booking Message',
                    'The package organizer sent you a message about your booking.',
                    'conversation',
                    null,
                    ['conversation_id' => $conversation->id]
                ));

                return;
            }

            if ((int) $sender->id === (int) $conversation->user_id) {
                $conversation->organizer?->notify(new SystemNotification(
                    'New Package Booking Message',
                    'A customer replied to your package-booking conversation.',
                    'conversation',
                    null,
                    ['conversation_id' => $conversation->id]
                ));

                return;
            }
        }

        $this->notifyActiveStaff(
            'New Visitor Message',
            "{$sender->first_name} {$sender->last_name} sent a new message in an E-PULAG conversation.",
            ['conversation_id' => $conversation->id]
        );
    }
}

                'string',

        $package = OrganizerPackage::with([
            'services',
        ])->findOrFail($validated['package_id']);

        /*
         * Walk-in Packages are processed exclusively by DENR staff.
         * They must not be booked through the public online endpoint.
         */
        if ($package->is_walk_in) {
            return response()->json([
                'message' => 'The Walk-in Package can only be processed by DENR staff and cannot be booked online.',
            ], 422);
        }

        if ($package->approval_status !== 'APPROVED') {
            return response()->json([
                'message' => 'This package is not yet approved for public booking.',
            ], 422);
        }

        if ($package->status !== 'ACTIVE') {
            return response()->json([
                'message' => 'This package is currently inactive.',
            ], 422);
        }

        if ($package->is_walk_in && ($validated['tourist_count'] !== 1 || count($validated['tourists']) !== 1)) {
            return response()->json(['message' => 'The DENR Individual Package allows one adult per booking.'], 422);
        }

        if ($package->is_walk_in && (int) $validated['tourists'][0]['age'] < 18) {
            return response()->json(['message' => 'The DENR Individual Package is currently available to adults aged 18 and above.'], 422);
        }

        if ($validated['tourist_count'] > $package->max_tourists) {
            return response()->json([
                'message' =>
                    'The number of tourists exceeds the maximum allowed for this package.',
            ], 422);
        }

        if (count($validated['tourists']) !== $validated['tourist_count']) {
            return response()->json([
                'message' =>
                    'Tourist count does not match the number of tourist records submitted.',
            ], 422);
        }

        if ($validated['tourist_count'] > 12) {
            return response()->json([
                'message' =>
                    'A package booking cannot exceed 12 tourists.',
            ], 422);
        }

        $trail = Trail::findOrFail($validated['trail_id']);

        if (!$trail->is_active) {
            return response()->json([
                'message' => 'The selected trail is inactive.',
            ], 422);
        }

        $hikingDate = $validated['hiking_date'];
        $today = now()->startOfDay();

        if ($hikingDate < $today->toDateString()) {
            return response()->json([
                'message' => 'The hiking date cannot be earlier than today.',
            ], 422);
        }

        /*
         * Create booking and tourist records
         * inside a database transaction.
         */
        $booking = DB::transaction(function () use (
            $validated,
            $request,
            $package
        ) {
            $capacityService = app(BookingCapacityService::class);
            $capacityService->lockForCapacityCheck();

            if (
                $summary['is_blocked'] ||
                $summary['current_capacity'] + $validated['tourist_count'] >
                    BookingCapacityService::MAXIMUM_CAPACITY
            ) {
                abort(response()->json([
                    'message' => 'There are not enough visitor slots remaining for this date.',
                ], 422));

                $package->is_walk_in &&
                $capacityService->getWalkInCount($validated['hiking_date']) + 1 >

                abort(response()->json([

                'denr_status' => $package->is_walk_in ? 'PENDING' : null,

                'remarks' => $validated['remarks'] ?? null,
            ]);

            foreach ($validated['tourists'] as $tourist) {
                $booking->tourists()->create([

        });

    /**
     * Stage 1:
     * Organizer approves or rejects an individual Tourist booking.
     *
     * Only the Organizer who owns the package can perform this action.
     */
    public function updateStatus(
        Request $request,
        PackageBooking $packageBooking
    ): JsonResponse {
        $user = $request->user();

        if ($user->role !== 'ORGANIZER') {
            return response()->json([
                'message' => 'Package booking not found.',
            ], 404);
        }

        // Verify that the Organizer owns this package.
        $ownsBooking = $packageBooking->package()
            ->where('organizer_id', $user->id)
            ->exists();

        if (!$ownsBooking) {
            return response()->json([
                'message' => 'Package booking not found.',
            ], 404);
        }

        $validated = $request->validate([
            'status' => [
                'required',
                'string',
                'in:APPROVED,REJECTED',
            ],
        ]);

        // Only pending bookings can be processed by the Organizer.
        if ($packageBooking->status !== 'PENDING') {
            return response()->json([
                'message' =>
                    'Only pending bookings can be approved or rejected.',
            ], 422);
        }

        $packageBooking->loadMissing([
            'customer',
            'package',
        ]);

        $customer = $packageBooking->customer;

        // Organizer decisions are only allowed for Tourist accounts.
        if (!$customer || $customer->role !== 'USER') {
            return response()->json([
                'message' =>
                    'This action is only available for tourist bookings.',
            ], 422);
        }

        $newStatus = $validated['status'];
        $approved = $newStatus === 'APPROVED';

        /*
         * The Organizer's decision is independent
         * for every package booking.
         */
        $packageBooking->update([
            'status' => $newStatus,

            // Only approved bookings proceed to DENR review.
            'denr_status' => $approved ? 'PENDING' : null,
        ]);

        $packageName = $packageBooking->package?->name
            ?? 'your package';

        /*
         * Notify the Tourist of the Organizer's decision.
         */
        $this->notifySafely(
            $customer,
            new SystemNotification(
                $approved
                    ? 'Package Booking Approved by Organizer'
                    : 'Package Booking Rejected by Organizer',
                $approved
                    ? "Your booking for {$packageName} has been approved by the Organizer and forwarded to DENR for final hike approval."
                    : "Your booking for {$packageName} has been rejected by the Organizer. It will not proceed to DENR review.",
                'package_booking',
                null,
                [
                    'booking_id' => $packageBooking->id,
                    'booking_number' => $packageBooking->booking_number,
                    'package_id' => $packageBooking->package_id,
                    'package_name' => $packageName,
                    'status' => $newStatus,
                    'denr_status' => $approved ? 'PENDING' : null,
                ]
            )
        );

        return response()->json([
            'message' => $approved
                ? 'Package booking approved and forwarded to DENR.'
                : 'Package booking rejected successfully.',
            'booking' => $packageBooking->fresh([
                'package.services',
                'package.organizer',
                'customer:id,first_name,last_name,email',
                'trail',
                'tourists',
            ]),
        ]);
    }

    public function updateDenrStatus(

        if (

    /**
     * Cancel a package booking.
     */
    public function destroy(
        Request $request,
        PackageBooking $packageBooking
    ): JsonResponse {
        if ($packageBooking->customer_id !== $request->user()->id) {
            return response()->json([
                'message' => 'Package booking not found.',
            ], 404);
        }

        if (
            in_array(
                $packageBooking->status,
                ['CANCELLED', 'COMPLETED'],
                true
            )
        ) {
            return response()->json([
                'message' =>
                    'This package booking can no longer be cancelled.',
            ], 422);
        }

        $packageBooking->loadMissing([
            'package.organizer:id,first_name,last_name,email',
        ]);

        $packageBooking->update([
            'status' => 'CANCELLED',
        ]);

        /*
         * Notify the package Organizer after cancellation.
         */
        $customer = $request->user();

        $customerName = trim(
            $customer->first_name . ' ' . $customer->last_name
        );

        $this->notifySafely(
            $packageBooking->package?->organizer,
            new SystemNotification(
                'Package Booking Cancelled',
                "{$customerName} cancelled package booking {$packageBooking->booking_number}.",
                'package_booking',
                null,
                [
                    'booking_id' => $packageBooking->id,
                    'booking_number' => $packageBooking->booking_number,
                    'customer_id' => $customer->id,
                    'package_id' => $packageBooking->package_id,
                    'package_name' => $packageBooking->package?->name,
                    'hiking_date' => $packageBooking->hiking_date,
                    'status' => 'CANCELLED',
                    'denr_status' => $packageBooking->denr_status,
                ]
            )
        );

        return response()->json([
            'message' => 'Package booking cancelled successfully.',
            'booking' => $packageBooking->fresh([
                'package',
                'trail',
                'tourists',
            ]),
        ]);
    }

    /**
     * Safely send a notification without rolling back
     * an already-completed booking operation.
     */
    private function notifySafely(
        ?User $recipient,
        SystemNotification $notification
    ): void {
        if (!$recipient) {
            return;
        }

        try {
            $recipient->notify($notification);
        } catch (Throwable $exception) {
            report($exception);
        }
    }

    /**
     * Generate a unique package booking number.
     */
    private function generateBookingNumber(): string
    {
        $year = now()->year;

        $latestBooking = PackageBooking::query()
            ->where('booking_number', 'like', "PB-{$year}-%")
            ->latest('id')
            ->first();

        if (!$latestBooking) {
            $nextNumber = 1;
        } else {
            $parts = explode('-', $latestBooking->booking_number);
            $nextNumber = ((int) end($parts)) + 1;
        }

        do {
            $bookingNumber = sprintf(
                'PB-%d-%04d',
                $year,
                $nextNumber
            );

            $exists = PackageBooking::query()
                ->where('booking_number', $bookingNumber)
                ->exists();

            if ($exists) {
                $nextNumber++;
            }
        } while ($exists);

        return $bookingNumber;
    }

            ->leftJoin('trails', 'package_bookings.trail_id', '=', 'trails.id')
            ->leftJoin('users', 'package_bookings.customer_id', '=', 'users.id')
            ->select([
                'package_bookings.booking_number',
                'package_bookings.package_id',
                'users.first_name',
                'users.last_name',
                'users.email',
                'package_bookings.hiking_date',
                'package_bookings.tourist_count',
                'package_bookings.status as organizer_status',
                'package_bookings.denr_status',
                'package_bookings.remarks',
                'trails.name as trail_name',
            ]);

        $this->applyDateFilter(
            $query,
            'package_bookings.hiking_date',
            $dateFrom,
            $dateTo
        );

        $rows = $query->orderBy('package_bookings.hiking_date')
            ->get()
            ->map(fn ($item) => (array) $item)
            ->all();

        return [
            [
                'Booking Number',
                'Package ID',
                'Customer First Name',
                'Customer Last Name',
                'Email',
                'Hiking Date',
                'Tourist Count',
                'Organizer Status',
                'DENR Status',
                'Remarks',
                'Trail',
            ],
            $rows,
        ];
    }

    /**
     * Visitor report across both direct reservations and package bookings.
     */
    private function visitorReport(
        ?string $dateFrom,
        ?string $dateTo
    ): array {
        $direct = DB::table('reservation_participants')
            ->join('reservations', 'reservation_participants.reservation_id', '=', 'reservations.id')
            ->select([
                'reservation_participants.full_name',
                'reservation_participants.age',
                'reservation_participants.gender',
                'reservation_participants.nationality',
                'reservation_participants.contact_number',
                'reservation_participants.address',
                'reservation_participants.emergency_contact',
                'reservation_participants.emergency_number',
                'reservations.reservation_number as booking_number',
                DB::raw("'Direct Reservation' as booking_type"),
                'reservations.hiking_date',
                'reservations.status as booking_status',
                DB::raw('NULL as denr_status'),
            ]);

        $this->applyDateFilter(
            $direct,
            'reservations.hiking_date',
            $dateFrom,
            $dateTo
        );

        $package = DB::table('package_booking_tourists')
            ->join(
                'package_bookings',
                'package_booking_tourists.package_booking_id',
                '=',
                'package_bookings.id'
            )
            ->select([
                'package_booking_tourists.full_name',
                'package_booking_tourists.age',
                'package_booking_tourists.gender',
                'package_booking_tourists.nationality',
                'package_booking_tourists.contact_number',
                'package_booking_tourists.address',
                'package_booking_tourists.emergency_contact',
                'package_booking_tourists.emergency_number',
                'package_bookings.booking_number',
                DB::raw("'Package Booking' as booking_type"),
                'package_bookings.hiking_date',
                'package_bookings.status as booking_status',
                'package_bookings.denr_status',
            ]);

        $this->applyDateFilter(
            $package,
            'package_bookings.hiking_date',
            $dateFrom,
            $dateTo
        );

        $rows = $direct
            ->unionAll($package)
            ->orderBy('hiking_date')
            ->get()
            ->map(fn ($item) => (array) $item)
            ->all();

        return [
            [
                'Visitor Name',
                'Age',
                'Gender',
                'Nationality',
                'Contact Number',
                'Address',
                'Emergency Contact',
                'Emergency Number',
                'Booking Number',
                'Booking Type',
                'Hiking Date',
                'Booking Status',
                'DENR Status',
            ],
            $rows,
        ];
    }

    /**
     * Daily visitor totals from bookings that still reserve park capacity.
     * Cancelled and rejected bookings are excluded, matching the booking capacity service.
     */
    private function dailyVisitorReport(
        ?string $dateFrom,
        ?string $dateTo
    ): array {
        $direct = DB::table('reservations')
            ->selectRaw('hiking_date, SUM(total_participants) as visitors')
            ->whereNotIn('status', ['CANCELLED', 'REJECTED']);

        $this->applyDateFilter(
            $direct,
            'hiking_date',
            $dateFrom,
            $dateTo
        );

        $direct = $direct->groupBy('hiking_date')->get();

        $packages = DB::table('package_bookings')
            ->selectRaw('hiking_date, SUM(tourist_count) as visitors')
            ->whereNotIn('status', ['CANCELLED', 'REJECTED'])
            ->where(function ($query) {
                $query->whereNull('denr_status')
                    ->orWhere('denr_status', '!=', 'REJECTED');
            });

        $this->applyDateFilter(
            $packages,
            'hiking_date',
            $dateFrom,
            $dateTo
        );

        $packages = $packages->groupBy('hiking_date')->get();

        $totals = [];

        foreach ($direct->concat($packages) as $item) {
            $date = (string) $item->hiking_date;

            if (!isset($totals[$date])) {
                $totals[$date] = 0;
            }

            $totals[$date] += (int) $item->visitors;
        }

        ksort($totals);

        $rows = [];

        foreach ($totals as $date => $visitors) {
            $rows[] = [
                'Hiking Date' => $date,
                'Total Visitors' => $visitors,
            ];
        }

        return [
            ['Hiking Date', 'Total Visitors'],
            $rows,
        ];
    }

    /**
     * Trail report.
     */
    private function trailReport(): array
    {
        $rows = DB::table('trails')
            ->select([
                'name',
                'description',
                'difficulty',
                'max_capacity',
                'is_active',
                'created_at',
            ])
            ->orderBy('name')
            ->get()
            ->map(fn ($item) => (array) $item)
            ->all();

        return [
            [
                'Trail Name',
                'Description',
                'Difficulty',
                'Maximum Capacity',
                'Active',
                'Created At',
            ],
            $rows,
        ];
    }

    /**
     * User account report. Passwords and tokens are never exported.
     */
    private function userReport(
        ?string $dateFrom,
        ?string $dateTo
    ): array {
        $query = DB::table('users')
            ->select([
                'first_name',
                'last_name',
                'email',
                'phone',
                'role',
                'status',
                'created_at',
            ]);

        $this->applyDateFilter(
            $query,
            'created_at',
            $dateFrom,
            $dateTo
        );

        $rows = $query->orderBy('created_at', 'desc')
            ->get()
            ->map(fn ($item) => (array) $item)
            ->all();

        return [
            [
                'First Name',
                'Last Name',
                'Email',
                'Phone',
                'Role',
                'Status',
                'Registered At',
            ],
            $rows,
        ];
    }

    /**
     * Apply optional date filters.
     */
    private function applyDateFilter(
        $query,
        string $column,
        ?string $dateFrom,
        ?string $dateTo
    ): void {
        if ($dateFrom) {
            $query->whereDate($column, '>=', $dateFrom);
        }

        if ($dateTo) {
            $query->whereDate($column, '<=', $dateTo);
        }
    }

    /**
     * Generate CSV download.
     */
    private function downloadCsv(
        string $filename,
        array $headings,
        array $rows
    ): StreamedResponse {
        return response()->streamDownload(function () use ($headings, $rows) {
            $output = fopen('php://output', 'w');

            // UTF-8 BOM for compatibility with Microsoft Excel.
            fwrite($output, "\xEF\xBB\xBF");

            fputcsv($output, $headings);

            foreach ($rows as $row) {
                fputcsv($output, array_values($row));
            }

            fclose($output);
        }, $filename . '.csv', [
            'Content-Type' => 'text/csv; charset=UTF-8',
        ]);
    }
