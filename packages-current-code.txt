"use client";

import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useMemo,
  useState,
  type ReactNode,
} from "react";
import { useSession } from "next-auth/react";

export type PackageStatus = "Active" | "Inactive";

export type ApiPackageService = {
  id: number;
  package_id: number;
  service: string;
};

export type ApiOrganizerPackage = {
  id: number;
  organizer_id: number;
  name: string;
  description: string;
  price: string | number;
  max_tourists: number;
  status: string;
  services: ApiPackageService[];
  created_at?: string;
  updated_at?: string;
};

export type OrganizerPackage = {
  id: number;
  name: string;
  description: string;
  price: number;
  services: string[];
  maxTourists: number;
  status: PackageStatus;
};

type PackageFormData = {
  name: string;
  description: string;
  price: number;
  services: string[];
};

type PackageContextValue = {
  packages: OrganizerPackage[];
  filteredPackages: OrganizerPackage[];

  search: string;
  statusFilter: string;

  loading: boolean;
  saving: boolean;

  togglingId: number | null;

  error: string;

  isFormOpen: boolean;
  isViewOpen: boolean;

  editingPackage: OrganizerPackage | null;
  viewingPackage: OrganizerPackage | null;

  totalPackages: number;
  activePackages: number;
  inactivePackages: number;

  setSearch: (value: string) => void;
  setStatusFilter: (value: string) => void;

  openCreate: () => void;
  openEdit: (packageItem: OrganizerPackage) => void;
  openView: (packageItem: OrganizerPackage) => void;

  closeForm: () => void;
  closeView: () => void;

  savePackage: (data: PackageFormData) => Promise<boolean>;
  toggleStatus: (packageItem: OrganizerPackage) => Promise<void>;

  refreshPackages: () => Promise<void>;
};

const PackageContext = createContext<PackageContextValue | undefined>(
  undefined
);

function formatApiStatus(status: string): PackageStatus {
  return status.toUpperCase() === "ACTIVE" ? "Active" : "Inactive";
}

function formatApiPackage(
  packageItem: ApiOrganizerPackage
): OrganizerPackage {
  return {
    id: packageItem.id,
    name: packageItem.name,
    description: packageItem.description,
    price: Number(packageItem.price),
    services: Array.isArray(packageItem.services)
      ? packageItem.services.map((service) => service.service)
      : [],
    maxTourists: Number(packageItem.max_tourists),
    status: formatApiStatus(packageItem.status),
  };
}

export function PackageProvider({
  children,
}: {
  children: ReactNode;
}) {
  const { data: session, status: sessionStatus } = useSession();

  const [packages, setPackages] = useState<OrganizerPackage[]>([]);
  const [search, setSearch] = useState("");
  const [statusFilter, setStatusFilter] = useState("All Status");

  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);
  const [togglingId, setTogglingId] = useState<number | null>(null);

  const [error, setError] = useState("");

  const [isFormOpen, setIsFormOpen] = useState(false);
  const [isViewOpen, setIsViewOpen] = useState(false);

  const [editingPackage, setEditingPackage] =
    useState<OrganizerPackage | null>(null);

  const [viewingPackage, setViewingPackage] =
    useState<OrganizerPackage | null>(null);

  const apiBaseUrl =
    process.env.NEXT_PUBLIC_LARAVEL_API_URL ||
    "http://localhost:8000";

  const accessToken = session?.user?.accessToken;

  const refreshPackages = useCallback(async () => {
    if (sessionStatus !== "authenticated" || !accessToken) {
      return;
    }

    setLoading(true);
    setError("");

    try {
      const response = await fetch(
        `${apiBaseUrl}/api/organizer/packages`,
        {
          headers: {
            Accept: "application/json",
            Authorization: `Bearer ${accessToken}`,
          },
          cache: "no-store",
        }
      );

      const data = await response.json();

      if (!response.ok) {
        throw new Error(
          data?.message || "Failed to load packages."
        );
      }

      const apiPackages: ApiOrganizerPackage[] =
        Array.isArray(data?.packages) ? data.packages : [];

      setPackages(apiPackages.map(formatApiPackage));
    } catch (err) {
      console.error("Failed to load organizer packages:", err);

      setPackages([]);

      setError(
        err instanceof Error
          ? err.message
          : "Failed to load packages."
      );
    } finally {
      setLoading(false);
    }
  }, [accessToken, apiBaseUrl, sessionStatus]);

  useEffect(() => {
    if (sessionStatus === "authenticated") {
      refreshPackages();
    }

    if (sessionStatus === "unauthenticated") {
      setPackages([]);
      setLoading(false);
    }
  }, [refreshPackages, sessionStatus]);

  const openCreate = useCallback(() => {
    setError("");
    setEditingPackage(null);
    setIsFormOpen(true);
  }, []);

  const openEdit = useCallback((packageItem: OrganizerPackage) => {
    setError("");
    setEditingPackage(packageItem);
    setIsFormOpen(true);
  }, []);

  const openView = useCallback((packageItem: OrganizerPackage) => {
    setError("");
    setViewingPackage(packageItem);
    setIsViewOpen(true);
  }, []);

  const closeForm = useCallback(() => {
    if (saving) {
      return;
    }

    setIsFormOpen(false);
    setEditingPackage(null);
    setError("");
  }, [saving]);

  const closeView = useCallback(() => {
    setIsViewOpen(false);
    setViewingPackage(null);
  }, []);

  const savePackage = useCallback(
    async (data: PackageFormData): Promise<boolean> => {
      if (!accessToken) {
        setError("You are not authenticated.");
        return false;
      }

      setSaving(true);
      setError("");

      try {
        const isEditing = Boolean(editingPackage);

        const url = isEditing
          ? `${apiBaseUrl}/api/organizer/packages/${editingPackage?.id}`
          : `${apiBaseUrl}/api/organizer/packages`;

        const method = isEditing ? "PATCH" : "POST";

        const response = await fetch(url, {
          method,
          headers: {
            "Content-Type": "application/json",
            Accept: "application/json",
            Authorization: `Bearer ${accessToken}`,
          },
          body: JSON.stringify({
            name: data.name,
            description: data.description,
            price: data.price,
            max_tourists: 12,
            services: data.services,
          }),
        });

        const responseData = await response.json();

        if (!response.ok) {
          if (responseData?.errors) {
            const firstError = Object.values(
              responseData.errors
            )
              .flat()
              .find(
                (message): message is string =>
                  typeof message === "string"
              );

            throw new Error(
              firstError ||
                responseData?.message ||
                "Failed to save package."
            );
          }

          throw new Error(
            responseData?.message ||
              "Failed to save package."
          );
        }

        await refreshPackages();

        setIsFormOpen(false);
        setEditingPackage(null);
        setError("");

        return true;
      } catch (err) {
        console.error("Failed to save package:", err);

        setError(
          err instanceof Error
            ? err.message
            : "Failed to save package."
        );

        return false;
      } finally {
        setSaving(false);
      }
    },
    [
      accessToken,
      apiBaseUrl,
      editingPackage,
      refreshPackages,
    ]
  );

  const toggleStatus = useCallback(
    async (packageItem: OrganizerPackage) => {
      if (!accessToken) {
        setError("You are not authenticated.");
        return;
      }

      setTogglingId(packageItem.id);
      setError("");

      try {
        const response = await fetch(
          `${apiBaseUrl}/api/organizer/packages/${packageItem.id}/status`,
          {
            method: "PATCH",
            headers: {
              Accept: "application/json",
              Authorization: `Bearer ${accessToken}`,
            },
          }
        );

        const data = await response.json();

        if (!response.ok) {
          throw new Error(
            data?.message ||
              "Failed to update package status."
          );
        }

        await refreshPackages();
      } catch (err) {
        console.error(
          "Failed to toggle package status:",
          err
        );

        setError(
          err instanceof Error
            ? err.message
            : "Failed to update package status."
        );
      } finally {
        setTogglingId(null);
      }
    },
    [accessToken, apiBaseUrl, refreshPackages]
  );

  const filteredPackages = useMemo(() => {
    const normalizedSearch = search.trim().toLowerCase();

    return packages.filter((packageItem) => {
      const matchesSearch =
        normalizedSearch === "" ||
        packageItem.name
          .toLowerCase()
          .includes(normalizedSearch) ||
        packageItem.description
          .toLowerCase()
          .includes(normalizedSearch) ||
        packageItem.services.some((service) =>
          service.toLowerCase().includes(normalizedSearch)
        );

      const matchesStatus =
        statusFilter === "All Status" ||
        packageItem.status === statusFilter;

      return matchesSearch && matchesStatus;
    });
  }, [packages, search, statusFilter]);

  const totalPackages = packages.length;

  const activePackages = packages.filter(
    (packageItem) => packageItem.status === "Active"
  ).length;

  const inactivePackages = packages.filter(
    (packageItem) => packageItem.status === "Inactive"
  ).length;

  const value = useMemo<PackageContextValue>(
    () => ({
      packages,
      filteredPackages,

      search,
      statusFilter,

      loading,
      saving,

      togglingId,

      error,

      isFormOpen,
      isViewOpen,

      editingPackage,
      viewingPackage,

      totalPackages,
      activePackages,
      inactivePackages,

      setSearch,
      setStatusFilter,

      openCreate,
      openEdit,
      openView,

      closeForm,
      closeView,

      savePackage,
      toggleStatus,

      refreshPackages,
    }),
    [
      packages,
      filteredPackages,
      search,
      statusFilter,
      loading,
      saving,
      togglingId,
      error,
      isFormOpen,
      isViewOpen,
      editingPackage,
      viewingPackage,
      totalPackages,
      activePackages,
      inactivePackages,
      openCreate,
      openEdit,
      openView,
      closeForm,
      closeView,
      savePackage,
      toggleStatus,
      refreshPackages,
    ]
  );

  return (
    <PackageContext.Provider value={value}>
      {children}
    </PackageContext.Provider>
  );
}

export function usePackages() {
  const context = useContext(PackageContext);

  if (!context) {
    throw new Error(
      "usePackages must be used inside PackageProvider."
    );
  }

  return context;
}

"use client";

import {
  useEffect,
  useState,
} from "react";
import { Users } from "lucide-react";

import Button from "@/components/ui/Button";
import Input from "@/components/ui/Input";
import Modal from "@/components/ui/Modal";

import type { OrganizerPackage } from "./PackageContext";

const SERVICE_OPTIONS = [
  "DENR Registration",
  "Transportation",
  "Guide / Tour",
  "Accommodation",
  "Food",
  "Other Services",
];

const MAX_TOURISTS = 12;

type PackageFormData = {
  name: string;
  description: string;
  price: number;
  services: string[];
};

type PackageFormModalProps = {
  open: boolean;
  packageItem: OrganizerPackage | null;
  saving: boolean;
  error: string;
  onClose: () => void;
  onSave: (
    data: PackageFormData
  ) => Promise<boolean>;
};

export default function PackageFormModal({
  open,
  packageItem,
  saving,
  error,
  onClose,
  onSave,
}: PackageFormModalProps) {
  const [name, setName] = useState("");
  const [description, setDescription] =
    useState("");
  const [price, setPrice] = useState("");
  const [services, setServices] = useState<
    string[]
  >([]);

  const isEditing = Boolean(packageItem);

  useEffect(() => {
    if (!open) {
      return;
    }

    if (packageItem) {
      setName(packageItem.name);
      setDescription(packageItem.description);
      setPrice(String(packageItem.price));
      setServices(packageItem.services);
    } else {
      setName("");
      setDescription("");
      setPrice("");
      setServices([]);
    }
  }, [open, packageItem]);

  const toggleService = (service: string) => {
    setServices((current) =>
      current.includes(service)
        ? current.filter(
            (item) => item !== service
          )
        : [...current, service]
    );
  };

  const handleSubmit = async (
    event: React.FormEvent<HTMLFormElement>
  ) => {
    event.preventDefault();

    const trimmedName = name.trim();
    const trimmedDescription =
      description.trim();
    const numericPrice = Number(price);

    if (!trimmedName) {
      return;
    }

    if (!trimmedDescription) {
      return;
    }

    if (
      !price ||
      Number.isNaN(numericPrice) ||
      numericPrice < 0
    ) {
      return;
    }

    if (services.length === 0) {
      return;
    }

    await onSave({
      name: trimmedName,
      description: trimmedDescription,
      price: numericPrice,
      services,
    });
  };

  return (
    <Modal
      open={open}
      onClose={onClose}
      title={
        isEditing
          ? "Edit Package"
          : "Create Package"
      }
    >
      <form
        onSubmit={handleSubmit}
        className="space-y-6"
      >
        {error && (
          <div className="rounded-2xl border border-red-200 bg-red-50 px-4 py-3 text-sm text-red-700 dark:border-red-900/50 dark:bg-red-950/20 dark:text-red-400">
            {error}
          </div>
        )}

        <div>
          <label
            htmlFor="package-name"
            className="mb-2 block text-sm font-medium text-zinc-700 dark:text-zinc-300"
          >
            Package Name
          </label>

          <Input
            id="package-name"
            value={name}
            onChange={(event) =>
              setName(event.target.value)
            }
            placeholder="e.g. Pulag Weekend Adventure"
            disabled={saving}
          />
        </div>

        <div>
          <label
            htmlFor="package-description"
            className="mb-2 block text-sm font-medium text-zinc-700 dark:text-zinc-300"
          >
            Description
          </label>

          <textarea
            id="package-description"
            value={description}
            onChange={(event) =>
              setDescription(event.target.value)
            }
            placeholder="Describe what your package offers..."
            disabled={saving}
            rows={4}
            className="w-full rounded-xl border border-zinc-300 bg-white px-3 py-2.5 text-sm text-zinc-900 outline-none transition placeholder:text-zinc-400 focus:border-amber-500 focus:ring-2 focus:ring-amber-500/20 disabled:cursor-not-allowed disabled:opacity-60 dark:border-zinc-700 dark:bg-zinc-900 dark:text-zinc-100"
          />
        </div>

        <div>
          <label
            htmlFor="package-price"
            className="mb-2 block text-sm font-medium text-zinc-700 dark:text-zinc-300"
          >
            Price per Tourist
          </label>

          <Input
            id="package-price"
            type="number"
            min="0"
            step="0.01"
            value={price}
            onChange={(event) =>
              setPrice(event.target.value)
            }
            placeholder="0.00"
            disabled={saving}
          />
        </div>

        <div>
          <div className="flex items-center justify-between gap-4">
            <div>
              <p className="text-sm font-medium text-zinc-700 dark:text-zinc-300">
                Tourist Capacity
              </p>

              <p className="mt-1 text-xs text-zinc-500 dark:text-zinc-400">
                Each individual package booking can
                have a maximum of 12 tourists.
              </p>
            </div>

            <div className="flex items-center gap-2 rounded-xl bg-zinc-100 px-3 py-2 text-sm font-medium text-zinc-700 dark:bg-zinc-900 dark:text-zinc-300">
              <Users className="h-4 w-4" />
              {MAX_TOURISTS}
            </div>
          </div>
        </div>

        <div>
          <p className="mb-3 text-sm font-medium text-zinc-700 dark:text-zinc-300">
            Included Services
          </p>

          <div className="grid gap-2 sm:grid-cols-2">
            {SERVICE_OPTIONS.map((service) => {
              const selected =
                services.includes(service);

              return (
                <label
                  key={service}
                  className={`flex cursor-pointer items-center gap-3 rounded-xl border p-3 transition ${
                    selected
                      ? "border-amber-400 bg-amber-50 dark:border-amber-700 dark:bg-amber-950/20"
                      : "border-zinc-200 bg-white hover:border-zinc-300 dark:border-zinc-800 dark:bg-zinc-950 dark:hover:border-zinc-700"
                  }`}
                >
                  <input
                    type="checkbox"
                    checked={selected}
                    onChange={() =>
                      toggleService(service)
                    }
                    disabled={saving}
                    className="h-4 w-4 rounded border-zinc-300 text-amber-600 focus:ring-amber-500"
                  />

                  <span className="text-sm text-zinc-700 dark:text-zinc-300">
                    {service}
                  </span>
                </label>
              );
            })}
          </div>
        </div>

        <div className="flex justify-end gap-3 border-t border-zinc-200 pt-5 dark:border-zinc-800">
          <Button
            type="button"
            variant="secondary"
            onClick={onClose}
            disabled={saving}
          >
            Cancel
          </Button>

          <Button
            type="submit"
            disabled={saving}
          >
            {saving
              ? "Saving..."
              : isEditing
                ? "Save Changes"
                : "Create Package"}
          </Button>
        </div>
      </form>
    </Modal>
  );
}

import type {

type PackageCardProps = {

    togglingId === packageItem.id;

            onClick={() => onView(packageItem)}
          >
            <Eye className="mr-1.5 h-4 w-4" />
            View
          </Button>

          <Button
            variant="secondary"
            onClick={() => onEdit(packageItem)}
          >
            <Pencil className="mr-1.5 h-4 w-4" />
            Edit
          </Button>

          <Button
            variant="secondary"
            disabled={isToggling}
            onClick={() =>
              onToggleStatus(packageItem)
            }
          >
            <Power className="mr-1.5 h-4 w-4" />

            {isToggling
              ? "..."
              : status === "Active"
                ? "Off"
                : "On"}
          </Button>
        </div>
      </div>
    </article>
  );
}

        </section>
      )}

      <PackageFormModal
        open={isFormOpen}
        packageItem={editingPackage}
        saving={saving}
        error={error}
        onClose={closeForm}
        onSave={savePackage}
      />

      <PackageViewModal
        open={isViewOpen}
        packageItem={viewingPackage}
        onClose={closeView}
      />
    </>
  );
}

<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use App\Models\OrganizerPackage;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class OrganizerPackageController extends Controller
{
    public function index(Request $request): JsonResponse
    {
        $packages = OrganizerPackage::with([
            'services',
        ])
            ->where(
                'organizer_id',
                $request->user()->id
            )
            ->latest()
            ->get();

        return response()->json([
            'packages' => $packages,
        ]);
    }

    public function show(
        Request $request,
        OrganizerPackage $package
    ): JsonResponse {
        if (
            $package->organizer_id !==
            $request->user()->id
        ) {
            return response()->json([
                'message' => 'Package not found.',
            ], 404);
        }

        $package->load([
            'services',
            'bookings',
        ]);

        return response()->json([
            'package' => $package,
        ]);
    }

    public function store(Request $request): JsonResponse
    {
        $validated = $request->validate([
            'name' => [
                'required',
                'string',
                'max:255',
            ],

            'description' => [
                'required',
                'string',
            ],

            'price' => [
                'required',
                'numeric',
                'min:0',
            ],

            'max_tourists' => [
                'required',
                'integer',
                'min:1',
                'max:12',
            ],

            'services' => [
                'required',
                'array',
                'min:1',
            ],

            'services.*' => [
                'required',
                'string',
                'max:255',
            ],
        ]);

        $package = DB::transaction(
            function () use (
                $validated,
                $request
            ) {
                $package = OrganizerPackage::create([
                    'organizer_id' =>
                        $request->user()->id,

                    'name' =>
                        $validated['name'],

                    'description' =>
                        $validated['description'],

                    'price' =>
                        $validated['price'],

                    'max_tourists' =>
                        $validated['max_tourists'],

                    'status' =>
                        'ACTIVE',
                ]);

                $package->services()
                    ->createMany(
                        collect(
                            $validated['services']
                        )
                            ->unique()
                            ->map(
                                fn ($service) => [
                                    'service' =>
                                        $service,
                                ]
                            )
                            ->values()
                            ->all()
                    );

                return $package;
            }
        );

        $package->load([
            'services',
        ]);

        return response()->json([
            'message' =>
                'Package created successfully.',

            'package' =>
                $package,
        ], 201);
    }

    public function update(
        Request $request,
        OrganizerPackage $package
    ): JsonResponse {
        if (
            $package->organizer_id !==
            $request->user()->id
        ) {
            return response()->json([
                'message' => 'Package not found.',
            ], 404);
        }

        $validated = $request->validate([
            'name' => [
                'required',
                'string',
                'max:255',
            ],

            'description' => [
                'required',
                'string',
            ],

            'price' => [
                'required',
                'numeric',
                'min:0',
            ],

            'max_tourists' => [
                'required',
                'integer',
                'min:1',
                'max:12',
            ],

            'services' => [
                'required',
                'array',
                'min:1',
            ],

            'services.*' => [
                'required',
                'string',
                'max:255',
            ],
        ]);

        DB::transaction(
            function () use (
                $validated,
                $package
            ) {
                $package->update([
                    'name' =>
                        $validated['name'],

                    'description' =>
                        $validated['description'],

                    'price' =>
                        $validated['price'],

                    'max_tourists' =>
                        $validated['max_tourists'],
                ]);

                $package->services()->delete();

                $package->services()
                    ->createMany(
                        collect(
                            $validated['services']
                        )
                            ->unique()
                            ->map(
                                fn ($service) => [
                                    'service' =>
                                        $service,
                                ]
                            )
                            ->values()
                            ->all()
                    );
            }
        );

        $package->load([
            'services',
        ]);

        return response()->json([
            'message' =>
                'Package updated successfully.',

            'package' =>
                $package,
        ]);
    }

    public function toggleStatus(
        Request $request,
        OrganizerPackage $package
    ): JsonResponse {
        if (
            $package->organizer_id !==
            $request->user()->id
        ) {
            return response()->json([
                'message' => 'Package not found.',
            ], 404);
        }

        $package->status =
            $package->status === 'ACTIVE'
                ? 'INACTIVE'
                : 'ACTIVE';

        $package->save();

        return response()->json([
            'message' =>
                $package->status === 'ACTIVE'
                    ? 'Package activated successfully.'
                    : 'Package deactivated successfully.',

            'package' => $package,
        ]);
    }
}
