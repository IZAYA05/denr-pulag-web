"use client";

import { useEffect, useState } from "react";

import {
  Mountain,
  MountainSnow,
  MountainIcon,
} from "lucide-react";

import { useDashboard } from "@/components/admin/DashboardClient";
import StatCard from "@/components/ui/StatCard";

type Trail = {
  id: number;
  is_active: boolean | number | string;
};

type ApiResponse = {
  trails?: Trail[];
  message?: string;
};

type TrailStatsData = {
  total: number;
  active: number;
  inactive: number;
};

const API_URL =
  process.env.NEXT_PUBLIC_LARAVEL_API_URL || "http://localhost:18000";

function isActiveTrail(value: Trail["is_active"]) {
  return value === true || value === 1 || value === "1";
}

export default function TrailStats() {
  const { accessToken } = useDashboard();

  const [trails, setTrails] = useState<TrailStatsData | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  useEffect(() => {
    const controller = new AbortController();

    async function fetchTrails() {
      if (!accessToken) {
        setTrails(null);
        setLoading(false);
        return;
      }

      try {
        setLoading(true);
        setError("");

        const response = await fetch(`${API_URL}/api/admin/trails`, {
          headers: {
            Accept: "application/json",
            Authorization: `Bearer ${accessToken}`,
          },
          cache: "no-store",
          signal: controller.signal,
        });

        const result: ApiResponse = await response.json();

        if (!response.ok) {
          throw new Error(
            result.message || "Failed to load trail statistics."
          );
        }

        const records = Array.isArray(result.trails)
          ? result.trails
          : [];

        const active = records.filter((trail) =>
          isActiveTrail(trail.is_active)
        ).length;

        setTrails({
          total: records.length,
          active,
          inactive: records.length - active,
        });
      } catch (err) {
        if (err instanceof Error && err.name === "AbortError") return;

        setTrails(null);

        setError(
          err instanceof Error
            ? err.message
            : "Unable to load trail statistics."
        );
      } finally {
        if (!controller.signal.aborted) {
          setLoading(false);
        }
      }
    }

    void fetchTrails();

    return () => controller.abort();
  }, [accessToken]);

  const stats = [
    {
      title: "Total Trails",
      value: trails?.total,
      subtitle: "All registered trails",
      icon: Mountain,
      color: "text-blue-600 dark:text-blue-400",
      bg: "bg-blue-100 dark:bg-blue-500/10",
    },
    {
      title: "Active Trails",
      value: trails?.active,
      subtitle: "Marked active in the database",
      icon: MountainSnow,
      color: "text-emerald-600 dark:text-emerald-400",
      bg: "bg-emerald-100 dark:bg-emerald-500/10",
    },
    {
      title: "Inactive Trails",
      value: trails?.inactive,
      subtitle: "Marked inactive in the database",
      icon: MountainIcon,
      color: "text-red-600 dark:text-red-400",
      bg: "bg-red-100 dark:bg-red-500/10",
    },
  ];

  return (
    <section className="space-y-3">
      {error && (
        <div
          role="alert"
          className="rounded-xl border border-red-200 bg-red-50 p-4 text-sm text-red-600 dark:border-red-900 dark:bg-red-950/30 dark:text-red-400"
        >
          {error}
        </div>
      )}

      <div className="grid gap-6 sm:grid-cols-2 xl:grid-cols-3">
        {stats.map((stat) => {
          const Icon = stat.icon;

          return (
            <StatCard
              key={stat.title}
              title={stat.title}
              value={
                loading || error || stat.value === undefined
                  ? "â€”"
                  : stat.value.toLocaleString()
              }
              subtitle={stat.subtitle}
              iconColor={stat.color}
              iconBg={stat.bg}
            >
              <Icon className="h-7 w-7" />
            </StatCard>
          );
        })}
      </div>
    </section>
  );
}

"use client";

import { useCallback, useEffect, useState } from "react";
import { RefreshCw, Mountain } from "lucide-react";

import { useDashboard } from "@/components/admin/DashboardClient";
import TrailCard, { Trail } from "./TrailCard";

type ApiResponse = {
  trails?: Trail[];
  message?: string;
};

const API_URL =
  process.env.NEXT_PUBLIC_LARAVEL_API_URL || "http://localhost:18000";

export default function TrailGrid() {
  const { accessToken } = useDashboard();

  const [trails, setTrails] = useState<Trail[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  const fetchTrails = useCallback(
    async (signal?: AbortSignal) => {
      if (!accessToken) {
        setTrails([]);
        setLoading(false);
        return;
      }

      try {
        setLoading(true);
        setError("");

        const response = await fetch(`${API_URL}/api/admin/trails`, {
          headers: {
            Accept: "application/json",
            Authorization: `Bearer ${accessToken}`,
          },
          cache: "no-store",
          signal,
        });

        const result: ApiResponse = await response.json();

        if (!response.ok) {
          throw new Error(
            result.message || "Failed to load trail records."
          );
        }

        if (!signal?.aborted) {
          setTrails(
            Array.isArray(result.trails) ? result.trails : []
          );
        }
      } catch (err) {
        if (err instanceof Error && err.name === "AbortError") {
          return;
        }

        console.error(err);

        setError(
          err instanceof Error
            ? err.message
            : "Unable to load trail records."
        );
      } finally {
        if (!signal?.aborted) {
          setLoading(false);
        }
      }
    },
    [accessToken]
  );

  useEffect(() => {
    const controller = new AbortController();

    void fetchTrails(controller.signal);

    return () => controller.abort();
  }, [fetchTrails]);

  return (
    <section className="space-y-5">
      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
        <div>
          <h2 className="text-lg font-semibold text-zinc-900 dark:text-white">
            Registered Trails
          </h2>

          <p className="mt-1 text-sm text-zinc-500 dark:text-zinc-400">
            Trail information retrieved from the database.
          </p>
        </div>

        <button
          type="button"
          onClick={() => void fetchTrails()}
          disabled={loading}
          className="inline-flex items-center justify-center gap-2 rounded-xl border border-zinc-200 bg-white px-4 py-2.5 text-sm font-semibold text-zinc-700 transition hover:bg-zinc-50 disabled:cursor-not-allowed disabled:opacity-50 dark:border-zinc-700 dark:bg-zinc-900 dark:text-zinc-200 dark:hover:bg-zinc-800"
        >
          <RefreshCw
            size={16}
            className={loading ? "animate-spin" : ""}
          />
          Refresh
        </button>
      </div>

      {loading ? (
        <div className="flex min-h-48 items-center justify-center rounded-2xl border border-zinc-200 bg-white dark:border-zinc-800 dark:bg-zinc-900">
          <div className="flex items-center gap-3 text-sm text-zinc-500">
            <RefreshCw size={18} className="animate-spin" />
            Loading trails...
          </div>
        </div>
      ) : error ? (
        <div
          role="alert"
          className="rounded-xl border border-red-200 bg-red-50 p-5 text-sm text-red-600 dark:border-red-900 dark:bg-red-950/30 dark:text-red-400"
        >
          <p>{error}</p>

          <button
            type="button"
            onClick={() => void fetchTrails()}
            className="mt-3 font-semibold underline"
          >
            Try again
          </button>
        </div>
      ) : trails.length === 0 ? (
        <div className="flex flex-col items-center justify-center rounded-2xl border border-dashed border-zinc-300 bg-white px-6 py-16 text-center dark:border-zinc-700 dark:bg-zinc-900">
          <div className="rounded-2xl bg-zinc-100 p-4 text-zinc-500 dark:bg-zinc-800 dark:text-zinc-400">
            <Mountain size={28} />
          </div>

          <h3 className="mt-4 font-semibold text-zinc-900 dark:text-white">
            No trails registered
          </h3>

          <p className="mt-2 max-w-sm text-sm text-zinc-500 dark:text-zinc-400">
            Trail records will appear here once they are added to the system.
          </p>
        </div>
      ) : (
        <>
          <div className="grid gap-6 md:grid-cols-2 xl:grid-cols-3">
            {trails.map((trail) => (
              <TrailCard key={trail.id} trail={trail} />
            ))}
          </div>

          <p className="text-xs text-zinc-500 dark:text-zinc-400">
            Showing {trails.length} registered trail records.
          </p>
        </>
      )}
    </section>
  );
}

"use client";

import { useEffect, useMemo, useState } from "react";
import { X } from "lucide-react";
import { useEmployeeDashboard } from "@/components/employee/EmployeeDashboardClient";

type Trail = {
  id: number;
  name: string;
  description?: string | null;
  difficulty?: string | null;
  image?: string | null;
  max_capacity?: number | null;
  is_active?: boolean | number | string | null;
  approved_reservations_count?: number | string | null;
  approved_package_bookings_count?: number | string | null;
};

type TrailsResponse = {
  trails?: Trail[];
  message?: string;
};

const API_URL =
  process.env.NEXT_PUBLIC_LARAVEL_API_URL || "http://localhost:18000";

function isTrailActive(value: Trail["is_active"]) {
  return value === true || value === 1 || value === "1";
}

function difficultyStyle(difficulty?: string | null) {
  switch ((difficulty || "").toLowerCase()) {
    case "easy":
      return "bg-emerald-100 text-emerald-800 dark:bg-emerald-900/40 dark:text-emerald-300";
    case "moderate":
    case "medium":
      return "bg-amber-100 text-amber-800 dark:bg-amber-900/40 dark:text-amber-300";
    case "difficult":
    case "hard":
    case "challenging":
      return "bg-red-100 text-red-800 dark:bg-red-900/40 dark:text-red-300";
    default:
      return "bg-gray-100 text-gray-700 dark:bg-gray-800 dark:text-gray-300";
  }
}

function TrailCard({
  trail,
  onView,
}: {
  trail: Trail;
  onView: (trail: Trail) => void;
}) {
  const active = isTrailActive(trail.is_active);
  const directBookings = Number(
    trail.approved_reservations_count ?? 0
  );
  const packageBookings = Number(
    trail.approved_package_bookings_count ?? 0
  );
  const totalBookings = directBookings + packageBookings;

  const imageUrl = trail.image
    ? trail.image.startsWith("http")
      ? trail.image
      : `${API_URL}/storage/${trail.image.replace(/^\/+/, "")}`
    : null;

  return (
    <article className="overflow-hidden rounded-2xl border border-gray-200 bg-white transition hover:border-emerald-300 hover:shadow-md dark:border-gray-800 dark:bg-gray-900 dark:hover:border-emerald-800">
      <div className="relative h-36 bg-gradient-to-br from-emerald-800 via-emerald-700 to-green-950">
        {imageUrl && (
          // eslint-disable-next-line @next/next/no-img-element
          <img
            src={imageUrl}
            alt={trail.name}
            className="h-full w-full object-cover"
            onError={(event) => {
              event.currentTarget.style.display = "none";
            }}
          />
        )}

        <span
          className={`absolute right-3 top-3 rounded-full px-3 py-1 text-xs font-semibold ${
            active
              ? "bg-emerald-100 text-emerald-800"
              : "bg-gray-200 text-gray-700"
          }`}
        >
          {active ? "Active" : "Inactive"}
        </span>
      </div>

      <div className="space-y-4 p-5">
        <div>
          <div className="flex flex-wrap items-start justify-between gap-2">
            <h2 className="text-lg font-bold text-gray-900 dark:text-white">
              {trail.name}
            </h2>

            {trail.difficulty && (
              <span
                className={`rounded-full px-2.5 py-1 text-xs font-semibold ${difficultyStyle(
                  trail.difficulty
                )}`}
              >
                {trail.difficulty}
              </span>
            )}
          </div>

          <p className="mt-2 line-clamp-3 min-h-[3.75rem] text-sm leading-5 text-gray-500 dark:text-gray-400">
            {trail.description || "No trail description provided."}
          </p>
        </div>

        <div className="grid grid-cols-2 gap-3">
          <div className="rounded-xl bg-gray-50 p-3 dark:bg-gray-800/70">
            <p className="text-xs text-gray-500 dark:text-gray-400">
              Max Capacity
            </p>
            <p className="mt-1 text-xl font-bold text-gray-900 dark:text-white">
              {trail.max_capacity ?? "â€”"}
            </p>
          </div>

          <div className="rounded-xl bg-gray-50 p-3 dark:bg-gray-800/70">
            <p className="text-xs text-gray-500 dark:text-gray-400">
              Approved Bookings
            </p>
            <p className="mt-1 text-xl font-bold text-gray-900 dark:text-white">
              {totalBookings.toLocaleString()}
            </p>
          </div>
        </div>

        <button
          type="button"
          onClick={() => onView(trail)}
          className="w-full rounded-xl border border-gray-200 px-4 py-2.5 text-sm font-semibold text-gray-700 transition hover:border-emerald-500 hover:text-emerald-700 dark:border-gray-700 dark:text-gray-200 dark:hover:text-emerald-400"
        >
          View Trail Details
        </button>
      </div>
    </article>
  );
}

function Detail({
  label,
  value,
}: {
  label: string;
  value: string | number;
}) {
  return (
    <div className="rounded-xl bg-gray-50 p-4 dark:bg-gray-800/70">
      <p className="text-xs font-medium uppercase tracking-wide text-gray-500 dark:text-gray-400">
        {label}
      </p>
      <p className="mt-1 break-words text-sm font-semibold text-gray-900 dark:text-white">
        {value}
      </p>
    </div>
  );
}

function SummaryCard({
  label,
  value,
}: {
  label: string;
  value: number;
}) {
  return (
    <div className="rounded-2xl border border-gray-200 bg-white p-5 dark:border-gray-800 dark:bg-gray-900">
      <p className="text-sm font-medium text-gray-500 dark:text-gray-400">
        {label}
      </p>
      <p className="mt-2 text-3xl font-bold tracking-tight text-gray-900 dark:text-white">
        {value.toLocaleString()}
      </p>
    </div>
  );
}

export default function EmployeeTrails() {
  const { accessToken } = useEmployeeDashboard();

  const [trails, setTrails] = useState<Trail[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");
  const [search, setSearch] = useState("");
  const [statusFilter, setStatusFilter] = useState("ALL");
  const [difficultyFilter, setDifficultyFilter] = useState("ALL");
  const [selectedTrail, setSelectedTrail] = useState<Trail | null>(null);

  useEffect(() => {
    if (!accessToken) {
      setTrails([]);
      setLoading(false);
      return;
    }

    const controller = new AbortController();

    async function loadTrails() {
      setLoading(true);
      setError("");

      try {
        const response = await fetch(`${API_URL}/api/employee/trails`, {
          headers: {
            Accept: "application/json",
            Authorization: `Bearer ${accessToken}`,
          },
          cache: "no-store",
          signal: controller.signal,
        });

        const data: TrailsResponse = await response.json();

        if (!response.ok) {
          throw new Error(data.message || "Unable to load trails.");
        }

        setTrails(Array.isArray(data.trails) ? data.trails : []);
      } catch (err) {
        if (err instanceof Error && err.name === "AbortError") return;

        setError(
          err instanceof Error
            ? err.message
            : "Unable to load trail records."
        );
      } finally {
        if (!controller.signal.aborted) setLoading(false);
      }
    }

    void loadTrails();

    return () => controller.abort();
  }, [accessToken]);

  const filteredTrails = useMemo(() => {
    const term = search.trim().toLowerCase();

    return trails.filter((trail) => {
      const matchesSearch =
        !term ||
        [trail.name, trail.description, trail.difficulty].some(
          (value) =>
            String(value ?? "").toLowerCase().includes(term)
        );

      const active = isTrailActive(trail.is_active);

      const matchesStatus =
        statusFilter === "ALL" ||
        (statusFilter === "ACTIVE" && active) ||
        (statusFilter === "INACTIVE" && !active);

      const matchesDifficulty =
        difficultyFilter === "ALL" ||
        (trail.difficulty || "").toLowerCase() ===
          difficultyFilter.toLowerCase();

      return matchesSearch && matchesStatus && matchesDifficulty;
    });
  }, [trails, search, statusFilter, difficultyFilter]);

  const activeCount = trails.filter((trail) =>
    isTrailActive(trail.is_active)
  ).length;

  const inactiveCount = trails.length - activeCount;

  const totalApprovedBookings = trails.reduce(
    (sum, trail) =>
      sum +
      Number(trail.approved_reservations_count ?? 0) +
      Number(trail.approved_package_bookings_count ?? 0),
    0
  );

  const difficulties = Array.from(
    new Set(
      trails
        .map((trail) => trail.difficulty)
        .filter((value): value is string => Boolean(value))
    )
  );

  return (
    <div className="space-y-6">
      <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4">
        <SummaryCard label="Total Trails" value={trails.length} />
        <SummaryCard label="Active Trails" value={activeCount} />
        <SummaryCard label="Inactive Trails" value={inactiveCount} />
        <SummaryCard
          label="Approved Bookings"
          value={totalApprovedBookings}
        />
      </div>

      <section className="rounded-2xl border border-gray-200 bg-white p-4 dark:border-gray-800 dark:bg-gray-900 sm:p-5">
        <div className="grid grid-cols-1 gap-3 md:grid-cols-3">
          <input
            type="search"
            value={search}
            onChange={(event) => setSearch(event.target.value)}
            placeholder="Search trails..."
            aria-label="Search trails"
            className="rounded-xl border border-gray-300 bg-white px-4 py-2.5 text-sm text-gray-900 outline-none focus:border-emerald-500 focus:ring-2 focus:ring-emerald-500/20 dark:border-gray-700 dark:bg-gray-950 dark:text-white"
          />

          <select
            value={statusFilter}
            onChange={(event) => setStatusFilter(event.target.value)}
            aria-label="Filter by trail status"
            className="rounded-xl border border-gray-300 bg-white px-4 py-2.5 text-sm text-gray-900 outline-none focus:border-emerald-500 dark:border-gray-700 dark:bg-gray-950 dark:text-white"
          >
            <option value="ALL">All Trail Statuses</option>
            <option value="ACTIVE">Active Trails</option>
            <option value="INACTIVE">Inactive Trails</option>
          </select>

          <select
            value={difficultyFilter}
            onChange={(event) => setDifficultyFilter(event.target.value)}
            aria-label="Filter by trail difficulty"
            className="rounded-xl border border-gray-300 bg-white px-4 py-2.5 text-sm text-gray-900 outline-none focus:border-emerald-500 dark:border-gray-700 dark:bg-gray-950 dark:text-white"
          >
            <option value="ALL">All Difficulty Levels</option>
            {difficulties.map((difficulty) => (
              <option key={difficulty} value={difficulty}>
                {difficulty}
              </option>
            ))}
          </select>
        </div>
      </section>

      {error && (
        <div
          role="alert"
          className="rounded-xl border border-red-200 bg-red-50 p-4 text-sm text-red-700 dark:border-red-900/50 dark:bg-red-950/30 dark:text-red-300"
        >
          {error}
        </div>
      )}

      {loading ? (
        <div className="grid grid-cols-1 gap-5 md:grid-cols-2 xl:grid-cols-3">
          {[1, 2, 3].map((item) => (
            <div
              key={item}
              className="h-80 animate-pulse rounded-2xl bg-gray-100 dark:bg-gray-800"
            />
          ))}
        </div>
      ) : filteredTrails.length === 0 ? (
        <div className="rounded-2xl border border-gray-200 bg-white px-6 py-16 text-center dark:border-gray-800 dark:bg-gray-900">
          <h2 className="font-semibold text-gray-900 dark:text-white">
            No trails found
          </h2>
          <p className="mt-1 text-sm text-gray-500 dark:text-gray-400">
            Try changing your search or filters.
          </p>
        </div>
      ) : (
        <div className="grid grid-cols-1 gap-5 md:grid-cols-2 xl:grid-cols-3">
          {filteredTrails.map((trail) => (
            <TrailCard
              key={trail.id}
              trail={trail}
              onView={setSelectedTrail}
            />
          ))}
        </div>
      )}

      {selectedTrail && (
        <div
          className="fixed inset-0 z-50 flex items-center justify-center bg-black/50 p-4 backdrop-blur-sm"
          onClick={() => setSelectedTrail(null)}
        >
          <section
            role="dialog"
            aria-modal="true"
            aria-labelledby="trail-details-title"
            className="max-h-[90vh] w-full max-w-xl overflow-y-auto rounded-2xl border border-gray-200 bg-white shadow-2xl dark:border-gray-800 dark:bg-gray-900"
            onClick={(event) => event.stopPropagation()}
          >
            <div className="flex items-start justify-between border-b border-gray-200 p-5 dark:border-gray-800">
              <div>
                <p className="text-sm font-medium text-emerald-600 dark:text-emerald-400">
                  Trail Details
                </p>
                <h2
                  id="trail-details-title"
                  className="mt-1 text-xl font-bold text-gray-900 dark:text-white"
                >
                  {selectedTrail.name}
                </h2>
              </div>

              <button
                type="button"
                onClick={() => setSelectedTrail(null)}
                aria-label="Close trail details"
                className="rounded-lg p-2 text-gray-500 hover:bg-gray-100 dark:hover:bg-gray-800"
              >
                <X size={20} />
              </button>
            </div>

            <div className="space-y-4 p-5">
              <p className="text-sm leading-6 text-gray-600 dark:text-gray-300">
                {selectedTrail.description || "No description provided."}
              </p>

              <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
                <Detail
                  label="Status"
                  value={
                    isTrailActive(selectedTrail.is_active)
                      ? "Active"
                      : "Inactive"
                  }
                />
                <Detail
                  label="Difficulty"
                  value={selectedTrail.difficulty || "Not specified"}
                />
                <Detail
                  label="Maximum Capacity"
                  value={selectedTrail.max_capacity ?? "Not specified"}
                />
                <Detail
                  label="Approved Direct Reservations"
                  value={Number(
                    selectedTrail.approved_reservations_count ?? 0
                  )}
                />
                <Detail
                  label="Approved Package Bookings"
                  value={Number(
                    selectedTrail.approved_package_bookings_count ?? 0
                  )}
                />
                <Detail
                  label="Total Approved Bookings"
                  value={
                    Number(
                      selectedTrail.approved_reservations_count ?? 0
                    ) +
                    Number(
                      selectedTrail.approved_package_bookings_count ?? 0
                    )
                  }
                />
              </div>

              <p className="rounded-xl bg-amber-50 p-3 text-xs leading-5 text-amber-800 dark:bg-amber-900/20 dark:text-amber-300">
                This employee view is read-only. Trail configuration and
                activation changes should be handled by authorized administrators.

            </div>

            <div className="flex justify-end border-t border-gray-200 p-5 dark:border-gray-800">
              <button
                type="button"
                onClick={() => setSelectedTrail(null)}
                className="rounded-xl bg-emerald-700 px-4 py-2.5 text-sm font-semibold text-white transition hover:bg-emerald-800"
              >
                Close
              </button>
            </div>

        </div>
      )}
    </div>
  );

<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use App\Models\PackageBooking;
use App\Models\Reservation;
use App\Models\Trail;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Carbon;

class EmployeeDashboardController extends Controller
{
    /**
     * Return the employee dashboard summary.
     *
     * Only bookings awaiting DENR review are included
     * in the employee's pending package booking count.
     */
    public function dashboard(): JsonResponse
    {
        $today = Carbon::today();

        $todayReservations = Reservation::query()
            ->whereDate('hiking_date', $today)
            ->withCount('participants')
            ->get();

        $todayPackageBookings = PackageBooking::query()
            ->whereDate('hiking_date', $today)
            ->withCount('tourists')
            ->get();

        $pendingDirect = Reservation::where('status', 'PENDING')
            ->count();

        $pendingPackages = PackageBooking::query()
            ->where('status', 'APPROVED')
            ->where('denr_status', 'PENDING')
            ->count();

        $confirmedDirectVisitors = $todayReservations
            ->where('status', 'APPROVED')
            ->sum('participants_count');

        $confirmedPackageVisitors = $todayPackageBookings
            ->filter(fn ($booking) =>
                $booking->status === 'APPROVED' &&
                $booking->denr_status === 'APPROVED'
            )
            ->sum('tourists_count');

        return response()->json([
            'date' => $today->toDateString(),

            'pending_direct_reservations' => $pendingDirect,
            'pending_package_bookings' => $pendingPackages,

            'pending_bookings' =>
                $pendingDirect + $pendingPackages,

            'confirmed_visitors_today' =>
                $confirmedDirectVisitors + $confirmedPackageVisitors,

            'trails' => [
                'active' => Trail::where('is_active', true)->count(),
                'inactive' => Trail::where('is_active', false)->count(),
            ],
        ]);
    }

    /**
     * Return direct reservations and package bookings for employees.
     */
    public function bookings(): JsonResponse
    {
        $reservations = Reservation::query()
            ->with(['user', 'trail', 'participants'])
            ->latest()
            ->get()
            ->map(fn (Reservation $reservation) => [
                'id' => $reservation->id,
                'type' => 'DIRECT',
                'booking_number' => $reservation->reservation_number,
                'group_leader_name' => $reservation->group_leader_name,
                'phone' => $reservation->group_leader_phone,
                'hiking_date' => $reservation->hiking_date,
                'total_visitors' => $reservation->participants->count(),
                'status' => $reservation->status,
                'remarks' => $reservation->remarks,
                'trail' => $reservation->trail,
                'user' => $reservation->user,
                'visitors' => $reservation->participants,
            ]);

        $packageBookings = PackageBooking::query()
            ->with([
                'package.organizer',
                'package.services',
                'customer',
                'trail',
                'tourists',
            ])
            ->latest()
            ->get()
            ->map(fn (PackageBooking $booking) => [
                'id' => $booking->id,
                'type' => 'PACKAGE',
                'booking_number' => $booking->booking_number,
                'hiking_date' => $booking->hiking_date,
                'total_visitors' => $booking->tourists->count(),

                // Stage 1: Organizer decision.
                'status' => $booking->status,

                // Stage 2: Final DENR decision.
                'denr_status' => $booking->denr_status,

                'remarks' => $booking->remarks,
                'trail' => $booking->trail,
                'customer' => $booking->customer,
                'package' => $booking->package,
                'visitors' => $booking->tourists,
            ]);

        return response()->json([
            'direct_reservations' => $reservations->values(),
            'package_bookings' => $packageBookings->values(),
        ]);
    }

    /**
     * Return visitor records from direct and package bookings.
     */
    public function visitors(): JsonResponse
    {
        $directVisitors = Reservation::query()
            ->with(['trail', 'participants'])
            ->latest()
            ->get()
            ->flatMap(function (Reservation $reservation) {
                return $reservation->participants->map(fn ($visitor) => [
                    'id' => $visitor->id,
                    'source' => 'DIRECT',
                    'booking_id' => $reservation->id,
                    'booking_number' => $reservation->reservation_number,
                    'hiking_date' => $reservation->hiking_date,
                    'booking_status' => $reservation->status,
                    'trail' => $reservation->trail?->name,
                    'full_name' => $visitor->full_name,
                    'age' => $visitor->age,
                    'gender' => $visitor->gender,
                    'nationality' => $visitor->nationality,
                    'contact_number' => $visitor->contact_number,
                    'address' => $visitor->address,
                    'emergency_contact' => $visitor->emergency_contact,
                    'emergency_number' => $visitor->emergency_number,
                ]);
            });

        $packageVisitors = PackageBooking::query()
            ->with(['trail', 'tourists'])
            ->latest()
            ->get()
            ->flatMap(function (PackageBooking $booking) {
                return $booking->tourists->map(fn ($visitor) => [
                    'id' => $visitor->id,
                    'source' => 'PACKAGE',
                    'booking_id' => $booking->id,
                    'booking_number' => $booking->booking_number,
                    'hiking_date' => $booking->hiking_date,

                    // Organizer decision.
                    'booking_status' => $booking->status,

                    // DENR final decision.
                    'denr_status' => $booking->denr_status,

                    'trail' => $booking->trail?->name,
                    'full_name' => $visitor->full_name,
                    'age' => $visitor->age,
                    'gender' => $visitor->gender,
                    'nationality' => $visitor->nationality,
                    'contact_number' => $visitor->contact_number,
                    'address' => $visitor->address,
                    'emergency_contact' => $visitor->emergency_contact,
                    'emergency_number' => $visitor->emergency_number,
                ]);
            });

        return response()->json([
            'visitors' => $directVisitors
                ->concat($packageVisitors)
                ->values(),
        ]);
    }

    /**
     * Return daily confirmed visitor totals for a requested date.
     */
    public function reports(Request $request): JsonResponse
    {
        $validated = $request->validate([
            'date' => ['nullable', 'date_format:Y-m-d'],
        ]);

        $date = Carbon::parse($validated['date'] ?? now()->toDateString())
            ->toDateString();

        $directReservations = Reservation::query()
            ->whereDate('hiking_date', $date)
            ->withCount('participants')
            ->get();

        $packageBookings = PackageBooking::query()
            ->whereDate('hiking_date', $date)
            ->withCount('tourists')
            ->get();

        $approvedDirect = $directReservations
            ->where('status', 'APPROVED');

        $awaitingOrganizer = $packageBookings
            ->where('status', 'PENDING');

        $awaitingDenr = $packageBookings
            ->filter(fn ($booking) =>
                $booking->status === 'APPROVED' &&
                $booking->denr_status === 'PENDING'
            );

        $approvedPackages = $packageBookings
            ->filter(fn ($booking) =>
                $booking->status === 'APPROVED' &&
                $booking->denr_status === 'APPROVED'
            );

        $rejectedByOrganizer = $packageBookings
            ->where('status', 'REJECTED');

        $rejectedByDenr = $packageBookings
            ->filter(fn ($booking) =>
                $booking->status === 'APPROVED' &&
                $booking->denr_status === 'REJECTED'
            );

        return response()->json([
            'date' => $date,

            'direct_reservations' => [
                'total' => $directReservations->count(),

                'pending' => $directReservations
                    ->where('status', 'PENDING')
                    ->count(),

                'approved' => $approvedDirect->count(),

                'rejected' => $directReservations
                    ->where('status', 'REJECTED')
                    ->count(),

                'confirmed_visitors' => $approvedDirect
                    ->sum('participants_count'),
            ],

            'package_bookings' => [
                'total' => $packageBookings->count(),

                'pending' =>
                    $awaitingOrganizer->count() + $awaitingDenr->count(),

                'awaiting_organizer' => $awaitingOrganizer->count(),

                'awaiting_denr' => $awaitingDenr->count(),

                'approved' => $approvedPackages->count(),

                'rejected' =>
                    $rejectedByOrganizer->count() +
                    $rejectedByDenr->count(),

                'rejected_by_organizer' =>
                    $rejectedByOrganizer->count(),

                'rejected_by_denr' =>
                    $rejectedByDenr->count(),

                'confirmed_visitors' => $approvedPackages
                    ->sum('tourists_count'),
            ],

            'confirmed_visitors' =>
                $approvedDirect->sum('participants_count')
                + $approvedPackages->sum('tourists_count'),
        ]);
    }

    /**
     * Return trail details and capacity configuration.
     */
    public function trails(): JsonResponse
    {
        return response()->json([
            'trails' => Trail::query()
                ->withCount([
                    'reservations as approved_reservations_count' => fn ($query) =>
                        $query->where('status', 'APPROVED'),

                    'packageBookings as approved_package_bookings_count' => fn ($query) =>
                        $query
                            ->where('status', 'APPROVED')
                            ->where('denr_status', 'APPROVED'),
                ])
                ->orderBy('name')
                ->get(),
        ]);
    }
